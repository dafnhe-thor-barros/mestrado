def conv_orbit(series, tol):
    ref = series[-1]
    for j in range(len(series)):
        if all(abs(x - ref) <= tol for x in series[j:]):
            return j + 1
    return len(series)


def bess_run(par, orb, mode="nominal", n_orb=N_ORB, dt=2.0, p_it_sun=None, p_pv=None):
    T = orb["T"]
    te = orb["t_ecl"]

    if mode == "falha":
        p_it_ecl = par["p_it_ecl_falha"]
        p_it_sol = par["p_it_nom"]
    else:
        p_it_ecl = orb["p_it_ecl"]
        p_it_sol = par["p_it_sun_nom"]

    if p_it_sun is not None:
        p_it_sol = p_it_sun
    if p_pv is None:
        p_pv = par["p_pv_eol"]

    eta_c = par["eta_conv"]  # ndab = 0.95
    p_ecl_bus = (p_it_ecl + par["p_aux"]) / eta_c
    surplus = max(0.0, p_pv - (p_it_sol + par["p_aux"]) / eta_c)

    n = int(round(n_orb * T / dt))
    soc = np.empty(n + 1)
    v = np.empty(n + 1)
    ib = np.empty(n + 1)
    soc[0] = 1.0

    E = par["e_bat_j"]
    r0 = par["r0"]
    nc = par["n_cells"]
    o0 = par["ocv0"]
    o1 = par["ocv1"]
    v_ov = par["v_ov"]
    v_uv = par["v_uv"]
    soc_req = par["soc_min_req"]
    i_cc_max = par["c_rate_chg"] * par["cap_ah"]
    eta_chg = par["eta_chg"]
    eta_dis = par["eta_dis"]

    trip_soc = None
    trip_uv = None
    collapse = None
    e_dis = 0.0
    e_chg = 0.0
    clip = 0.0

    for k in range(n):
        t = k * dt
        s = soc[k]
        ocv = nc * (o0 + o1 * s)

        # Descarga a potência constante no eclipse
        if (t % T) < te:
            disc = ocv * ocv - 4.0 * r0 * p_ecl_bus
            if disc <= 0.0:
                if collapse is None:
                    collapse = t / 60.0
                i_k = ocv / (2.0 * r0)
                v_k = 0.5 * ocv
            else:
                i_k = (ocv - math.sqrt(disc)) / (2.0 * r0)
                v_k = ocv - i_k * r0
            ds = -p_ecl_bus / (E * eta_dis) * dt
            e_dis += p_ecl_bus * dt
            ib_k = i_k
        # Recarga CC-CV no período de insolação
        else:
            if s >= 1.0:
                i_c = 0.0
                v_k = ocv
            else:
                i_pow = (-ocv + math.sqrt(ocv * ocv + 4.0 * r0 * surplus)) / (2.0 * r0)
                i_c = min(i_pow, i_cc_max)
                v_k = ocv + i_c * r0
                if v_k > v_ov:
                    i_c = max(0.0, (v_ov - ocv) / r0)
                    v_k = v_ov
            ds = i_c * v_k * eta_chg / E * dt
            e_chg += i_c * v_k * dt
            ib_k = -i_c

        s_new = s + ds
        if s_new > 1.0:
            clip += (s_new - 1.0) * E
            s_new = 1.0
        elif s_new < 0.0:
            clip += s_new * E
            s_new = 0.0

        soc[k + 1] = s_new
        v[k] = v_k
        ib[k] = ib_k

        if trip_soc is None and s_new < soc_req:
            trip_soc = (k + 1) * dt / 60.0
        if trip_uv is None and v_k < v_uv:
            trip_uv = t / 60.0

    v[n] = v[n - 1]
    ib[n] = ib[n - 1]

    closure = (soc[-1] - 1.0) * E - (e_chg * eta_chg - e_dis / eta_dis - clip)
    t_arr = np.arange(n + 1) * dt
    idx = np.minimum((t_arr / T).astype(int), n_orb - 1)
    bounds = np.searchsorted(idx, np.arange(n_orb + 1))
    pw = ib * v
    pw[-1] = 0.0

    orbits = []
    for j in range(n_orb):
        a = int(bounds[j])
        b = int(bounds[j + 1])
        if b - a < 2:
            continue
        sl = slice(a, b)
        p_j = pw[sl]
        orbits.append({
            "j": j + 1,
            "soc_min": float(soc[sl].min() * 100.0),
            "soc_end": float(soc[b - 1] * 100.0),
            "v_min": float(v[sl].min()),
            "v_max": float(v[sl].max()),
            "i_dis_max": float(ib[sl].max()),
            "i_chg_max": float(-ib[sl].min()),
            "e_dis_kwh": float(p_j[p_j > 0].sum() * dt / 3.6e6),
            "e_chg_kwh": float(-p_j[p_j < 0].sum() * dt / 3.6e6)
        })

    st = max(1, int(round(10.0 / dt)))
    return {
        "orbit": orb["name"], "T": T, "mode": mode,
        "p_it_sol": p_it_sol, "p_pv": p_pv, "orbits": orbits,
        "trip_soc_min": trip_soc, "trip_uv_min": trip_uv,
        "collapse_min": collapse, "closure_rel": float(closure / E),
        "t_h": t_arr[::st] / 3600.0, "soc": soc[::st] * 100.0,
        "v": v[::st], "i": ib[::st]
    }


def bess_recovers(res, tol=99.9):
    return (
        res["trip_soc_min"] is None and
        res["trip_uv_min"] is None and
        res["collapse_min"] is None and
        all(o["soc_end"] >= tol for o in res["orbits"])
    )


def p_it_sun_max(par, orb, p_pv, n_orb=N_ORB, dt=2.0):
    hi = min(par["p_it_nom"], p_pv * par["eta_conv"] - par["p_aux"])
    r_hi = bess_run(par, orb, "nominal", 8, 20.0, p_it_sun=hi, p_pv=p_pv)
    if bess_recovers(r_hi):
        lo = hi
    else:
        lo = 0.0
        for _ in range(16):
            mid = 0.5 * (lo + hi)
            r = bess_run(par, orb, "nominal", 8, 20.0, p_it_sun=mid, p_pv=p_pv)
            if bess_recovers(r):
                lo = mid
            else:
                hi = mid
    ps = lo
    r = None
    for _ in range(60):
        r = bess_run(par, orb, "nominal", n_orb, dt, p_it_sun=ps, p_pv=p_pv)
        if bess_recovers(r):
            return ps, r
        ps -= 100.0
        if ps < 0.0:
            break
    return 0.0, r


def short_circuit(par, t_trip, L, C, clamp=True, battery=False, dt=10e-9):
    R = par["r_line"] + par["r_sc"]
    vcl = par["v_clamp"]
    vb = par["v_bat_sc"]
    r0b = par["r0_bat_sc"]
    Lb = par["l_bat"]
    rds = par["r_ds_on"]

    n_max = int((t_trip + 600e-6) / dt)
    iL = 0.0
    ib = 0.0
    vc = par["v_nom"]
    i2t = 0.0
    i2t_pre = None
    i_trip = 0.0
    vc_trip = vc
    e_tvs = 0.0
    ipk = 0.0
    vsw_max = 0.0
    t_end_cond = None
    extinguished = False

    for k in range(n_max):
        t = k * dt
        if t < t_trip:
            vs = iL * rds
            iL = iL + (vc - R * iL) / L * dt
        else:
            if i2t_pre is None:
                i2t_pre = i2t
                i_trip = iL
                vc_trip = vc
            if iL <= 1e-3:
                t_end_cond = t
                extinguished = True
                break
            if clamp:
                vs = vcl
                iL = max(0.0, iL + (vc - vcl - R * iL) / L * dt)
                e_tvs += vcl * iL * dt
            else:
                vs = vc + L * iL / par["tau_ext"] - R * iL
                iL = max(0.0, iL - iL / par["tau_ext"] * dt)

        if battery:
            ib = ib + (vb - r0b * ib - vc) / Lb * dt
            vc = vc + (ib - iL) / C * dt
        else:
            vc = vc - iL / C * dt

        i2t += iL * iL * dt
        if iL > ipk:
            ipk = iL
        if vs > vsw_max:
            vsw_max = vs

    if i2t_pre is None:
        i2t_pre = i2t
    if t_end_cond is None:
        t_end_cond = n_max * dt

    return {
        "t_trip_us": t_trip * 1e6, "ipk_kA": ipk / 1e3, "i_trip_kA": i_trip / 1e3,
        "vc_trip_V": vc_trip, "i2t_total": i2t, "i2t_pre_trip": i2t_pre,
        "e_tvs_J": e_tvs, "v_sw_max_V": vsw_max,
        "t_ext_us": (t_end_cond - t_trip) * 1e6, "extinguished": extinguished
    }


def chassis_run(par, eta, g_th, t_sink, n_orb=N_ORB, dt=2.0):
    t_ref = par["t_ref_chassi"]
    p_conv = par["p_node"] * (1.0 - eta)
    q_o = par["q_oisl"]
    c = par["c_th_chassi"]

    n = int(round(n_orb * t_ref / dt))
    T = np.empty(n + 1)
    T[0] = par["t_chassi0"]

    for k in range(n):
        tc = (k * dt) % 1800.0
        duty = 0.4 + (0.6 if tc < 1200.0 else 0.0)
        q = p_conv * duty + q_o
        T[k + 1] = T[k] + (q - g_th * (T[k] - t_sink)) / c * dt

    idx = np.minimum((np.arange(n + 1) * dt / t_ref).astype(int), n_orb - 1)
    bounds = np.searchsorted(idx, np.arange(n_orb + 1))
    tmax_orb = [float(T[bounds[j]:bounds[j + 1]].max()) for j in range(n_orb)]
    q_avg = p_conv * 0.8 + q_o

    return {
        "p_loss_pk": p_conv + q_o,
        "t_max_last": tmax_orb[-1],
        "t_eq_medio": t_sink + q_avg / g_th,
        "margin": par["t_lim_chassi"] - tmax_orb[-1],
        "status": "OK" if (par["t_lim_chassi"] - tmax_orb[-1]) >= 0 else "VIOLA"
    }

def main():
    t0 = time.time()
    os.makedirs(OUT_DIR, exist_ok=True)
    print("=" * 80)
    print("SIMULAÇÃO 3.1: DINÂMICA DO BARRAMENTO E BESS (COM OISL CONTÍNUO = 440 W)")
    print("=" * 80)

    # 1. Simulação BESS
    bess_resumo = []
    bess_resultados_completos = []
    for orb in ORBITS:
        r_sus = p_it_sun_max(par=PAR, orb=orb, p_pv=PAR["p_pv_eol"])[1]
        cases = [
            ("nominal_32kW_script", bess_run(PAR, orb, "nominal", p_it_sun=PAR["p_it_sun_nom"])),
            ("nominal_40kW_cap4", bess_run(PAR, orb, "nominal", p_it_sun=PAR["p_it_nom"])),
            ("nominal_Pmax_sustentavel", r_sus),
            ("falha_BAT_F01", bess_run(PAR, orb, "falha"))
        ]
        for cname, r in cases:
            ob = r["orbits"]
            ok = (
                r["trip_soc_min"] is None and
                r["trip_uv_min"] is None and
                min(o["soc_min"] for o in ob) >= 20.0
            )
            bess_resumo.append({
                "caso": cname, "orbita": orb["name"],
                "p_it_sol_kW": r["p_it_sol"] / 1e3,
                "soc_min_pct": ob[-1]["soc_min"],
                "v_min_V": ob[-1]["v_min"],
                "status": "OK" if ok else "VIOLA"
            })
            bess_resultados_completos.append(r)

    # 2. Resumo da Carga Máxima de TI Sustentável
    pmax_resumo = []
    for orb in ORBITS:
        for pvn, pvv in PV_SCEN:
            ps, rr = p_it_sun_max(PAR, orb, pvv)
            pmax_resumo.append({
                "orbita": orb["name"], "cenario_pv": pvn,
                "p_pv_kW": pvv / 1e3, "p_it_sun_max_kW": ps / 1e3,
                "folga_kW": (ps - PAR["p_it_nom"]) / 1e3
            })

    print("\n--- RESUMO BESS MULTI-ÓRBITA (20 Órbitas | Cargas Contínuas = 1000 W) ---")
    for r in bess_resumo:
        print("%-26s %-12s TI_sol=%4.1f kW | SoCmin=%5.2f%% | Vmin=%5.1f V | %s" % (
            r["caso"], r["orbita"], r["p_it_sol_kW"], r["soc_min_pct"], r["v_min_V"], r["status"]))

    print("\n--- CARGA MÁXIMA DE TI EM SOL (Pmax Sustentável) ---")
    for r in pmax_resumo:
        print("%-12s %-18s PV=%5.2f kW -> TI_sol_max=%5.2f kW (Folga: %5.2f kW)" % (
            r["orbita"], r["cenario_pv"], r["p_pv_kW"], r["p_it_sun_max_kW"], r["folga_kW"]))

    # Chamada da função para salvar os gráficos em PNG na pasta saida_3_1
    gerar_graficos(bess_resumo_completo := bess_resultados_completos)

    print(f"\nSimulação 3.1 executada em {time.time() - t0:.2f} s. Saída em ./{OUT_DIR}/")


if __name__ == "__main__":
    main()

def sub_params(par):
    """Calcula as correntes e tensões de saturação para cada uma das 3 subcélulas."""
    sc = par["g_am0"] / par["g_ds"]
    isc = par["jsc"] * par["a_cm2"] * sc
    return {
        "iph0": np.array(par["ratio"]) * isc,
        "diph": np.array(par["w_dph"]) * par["djsc"] * par["a_cm2"] * sc,
        "voc0": np.array(par["voc_sub"]),
        "dvoc": np.array(par["dvoc_sub"]),
        "n": np.array(par["n_sub"]),
        "rsh": np.array(par["rsh"]),
        "isc_cat": isc,
        "imp_cat": par["jmp"] * par["a_cm2"] * sc
    }


def v_sub(ic, iph, i0, vth, rsh, voc):
    """Resolve a tensão interna de uma subcélula pelo método de bissecção/Newton."""
    lo = np.zeros_like(ic)
    hi = np.full_like(ic, voc * 1.6)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        f = iph - i0 * (np.exp(np.minimum(mid / vth, 700.0)) - 1.0) - mid / rsh - ic
        pos = f > 0
        lo = np.where(pos, mid, lo)
        hi = np.where(pos, hi, mid)
    return 0.5 * (lo + hi)


def cell_iv(sp, rs, T, n=3000):
    """Calcula a curva I-V completa somando as 3 junções em série: Vcel = V1 + V2 + V3 - I*Rs."""
    dT = T - 28.0
    TK = T + 273.15
    iph = sp["iph0"] + sp["diph"] * dT
    voc = sp["voc0"] + sp["dvoc"] * dT
    vth = sp["n"] * K_B * TK / Q_E
    i0 = (iph - voc / sp["rsh"]) / (np.exp(voc / vth) - 1.0)

    ic = np.linspace(0.0, iph.min() * 0.9999, n)
    vs = np.zeros(n)
    for j in range(3):
        vs = vs + v_sub(ic, iph[j], i0[j], vth[j], sp["rsh"][j], voc[j])
    return ic, np.maximum(0.0, vs - ic * rs), float(voc.sum())


def mpp_cell(sp, rs, T):
    """Encontra o Ponto de Máxima Potência (MPP) unitário da célula a dada temperatura."""
    ic, v, voc = cell_iv(sp, rs, T)
    p = v * ic
    k = int(np.argmax(p))
    return {"vmp": float(v[k]), "imp": float(ic[k]), "pmp": float(p[k]), "voc": voc}


def calib_rs(sp, vmp_cat):
    """Calibra a resistência série interna para casar com a tensão de catálogo a 28°C."""
    lo, hi = 0.0, 0.3
    for _ in range(30):
        mid = 0.5 * (lo + hi)
        if mpp_cell(sp, mid, 28.0)["vmp"] > vmp_cat:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def array_curve(sp, rs, T, ns, npar):
    """Escala a curva da célula unitária para toda a matriz (Ns células em série, Np strings)."""
    ic, v, voc = cell_iv(sp, rs, T)
    return ns * v[::-1], npar * ic[::-1]


def bus_power(par, sp, rs, T, vbus_bat, ret):
    """Calcula a potência EOL entregue ao barramento HVDC pós-Buck com eta_buck = 0.91."""
    ic, v, voc = cell_iv(sp, rs, T)
    p = v * ic
    k = int(np.argmax(p))
    ns = par["ns"]

    # Restrição de saturação de razão cíclica D <= 0.90
    vmin_cell = (vbus_bat / par["d_max"]) / ns
    vref = max(v[k], vmin_cell)
    order = np.argsort(v)
    pat = float(np.interp(vref, v[order], p[order]))

    # Aplicação da eficiência do Buck ajustada para 91,0%
    p_str_bus = ns * pat * ret * par["eta_buck"]
    return {
        "vref_str": ns * vref,
        "p_mpp_str_W": ns * p[k],
        "p_bus_str_W": p_str_bus,
        "fracao_mpp": pat / float(p[k]),
        "np_req": math.ceil(par["p_alvo_bus"] / p_str_bus)
    }


def hotspot(par, pdiss, faces, q_amb=0.0, dt=0.01, tend=240.0):
    """Calcula a evolução térmica no vácuo de uma célula sombreada via Stefan-Boltzmann."""
    a = par["a_cm2"] * 1e-4
    T = par["hs_t0"] + 273.15
    t_lim = None
    tl = par["hs_tlim"] + 273.15

    for k in range(int(tend / dt)):
        q = faces * par["hs_eps"] * SIG * a * (T ** 4 - 3.0 ** 4)
        T = T + (pdiss + q_amb - q) * dt / par["hs_cth"]
        if t_lim is None and T >= tl:
            t_lim = (k + 1) * dt
    teq = ((pdiss + q_amb) / (faces * par["hs_eps"] * SIG * a) + 3.0 ** 4) ** 0.25 - 273.15
    return t_lim, teq, T - 273.15


def ddd_ret(par, cd, rate, yrs=None):
    """Calcula a retenção de potência pelo modelo de Dose de Deslocamento (DDD)."""
    if yrs is None:
        yrs = par["anos"]
    return (1.0 - cd * math.log10(1.0 + yrs * rate / par["phi_x"])) * (1.0 - par["dp_env"] * yrs / 10.0)


def run_32(par):
    out = {}
    sp = sub_params(par)
    rs_cal = calib_rs(sp, par["vmp_cat"])
    rs = rs_cal if par["rs_modo"] == "calibrado" else par["rs_texto"]
    ns, npar = par["ns"], par["np"]
    out["rs"] = rs
    out["rs_cal"] = rs_cal

    # 1. Curvas I-V da matriz
    out["iv"] = []
    for T in par["temps"]:
        m = mpp_cell(sp, rs, T)
        out["iv"].append({
            "T": T, "vmp_cel": m["vmp"], "vmp_str": ns * m["vmp"],
            "voc_str": ns * m["voc"], "imp_arr": npar * m["imp"],
            "pmp_kW": ns * npar * m["pmp"] / 1e3,
            "d_req_400": par["vbus"] / (ns * m["vmp"])
        })

    # 2. Degradação DDD
    out["ddd"] = []
    for cd in par["cd"]:
        for rate, alt in zip(par["fluencia"], par["alt_km"]):
            out["ddd"].append({"cd": cd, "alt_km": alt, "ret_pct": 100.0 * ddd_ret(par, cd, rate)})

    # 3. Potência entregue ao barramento HVDC EOL em SSO 700 km
    ret700 = ddd_ret(par, par["cd"][-1], par["fluencia"][1])
    out["bus"] = []
    for T in (28.0, 65.0, 85.0):
        for vb in (par["vbus"], par["v_ov"]):
            b = bus_power(par, sp, rs, T, vb, ret700)
            out["bus"].append({
                "T": T, "v_bat": vb, "v_ref_str": b["vref_str"],
                "p_mpp_BOL_kW": b["p_mpp_str_W"] * npar / 1e3,
                "p_bus_EOL_kW": b["p_bus_str_W"] * npar / 1e3,
                "fracao_mpp": b["fracao_mpp"], "np_req_alvo": b["np_req"]
            })

    # 4. Hot spot com e sem diodo de bypass
    out["hot"] = []
    imp = sp["imp_cat"]
    for nm, pd, f, qa in (
        ("sem_bypass_2faces", par["hs_v_rev"] * imp, 2, 0.0),
        ("sem_bypass_1face", par["hs_v_rev"] * imp, 1, 0.0),
        ("com_bypass_2faces", par["hs_v_byp"] * imp, 2, 0.0),
        ("com_bypass_2faces_q_amb", par["hs_v_byp"] * imp, 2, par["hs_q_amb"])
    ):
        t_l, teq, t_end = hotspot(par, pd, f, qa)
        out["hot"].append({"caso": nm, "p_diss_W": pd, "t_180_s": t_l, "t_eq_C": teq, "t_240s_C": t_end})

    return out


def write_csv(path, rows):
    if not rows:
        return
    keys = list(rows[0].keys())
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=keys, delimiter=";")
        w.writeheader()
        for r in rows:
            w.writerow({k: ("%.6g" % v if isinstance(v, float) else v) for k, v in r.items()})


def main():
    t0 = time.time()
    os.makedirs(OUT_DIR, exist_ok=True)
    R = run_32(PAR)

    for nome in ("iv", "ddd", "bus", "hot"):
        write_csv(os.path.join(OUT_DIR, nome + ".csv"), R[nome])

    print("=== POTENCIA NO BARRAMENTO (EOL SSO 700 km | nbuck = 0.91) ===")
    for r in R["bus"]:
        print("T=%4.0f C | Vbat=%5.1f V | Vref=%6.1f V | P_BOL=%5.2f kW | P_EOL_bus=%6.2f kW | Np_req=%d" % (
            r["T"], r["v_bat"], r["v_ref_str"], r["p_mpp_BOL_kW"], r["p_bus_EOL_kW"], r["np_req_alvo"]))
    print(f"\nSimulação concluída em {time.time() - t0:.2f} s. Saída em ./{OUT_DIR}/")


if __name__ == "__main__":
    main()

# Constantes Físicas Fundamentais
Q_E, K_B = 1.602176634e-19, 1.380649e-23
EA_K = 0.32 * Q_E / K_B                 # Energia de ativação de Arrhenius [K] (0,32 eV)
NS = 108                                # Células em série no pack Li-ion NMC
V_UV, V_OV = 324.0, 442.8               # Limiares de corte: UVLO (3,0 V) e CV (4,10 V)

# Parâmetros Eletroquímicos do Modelo DP de 2ª Ordem
RP1, CP1 = 18e-3, 3200.0                # Ramo de difusão rápida: tau1 = 57,6 s
RP2, CP2 = 24e-3, 22000.0               # Ramo de difusão lenta: tau2 = 528 s
R0_REF = 85e-3                          # Resistência ôhmica degradada em EOL a 20 °C [Ohm]
R_DIST = 14e-3                          # Resistência da cablagem de distribuição [Ohm]
T_REF_K = 293.15                        # 20 °C em Kelvin

# Acoplamento Térmico Condutivo Corrigido
C_BAT = 54360.0                         # Capacitância térmica do pack (60 kg NMC) [J/K]
RTH_COND = 0.14                         # Resistência condutiva estrutural [K/W]
T_CH = 15.0                             # Temperatura média do chassi de ancoragem [°C]
P_HEATER, T_ON, T_OFF = 300.0, 10.0, 15.0

# Dinâmica Orbital SSO (700 km)
T_ORB, T_ECL = 5926.0, 1128.0            # Período = 98,77 min; Eclipse sazonal máx = 18,8 min
ETA_DAB = 0.95                          # Eficiência do conversor de distribuição (ndab = 0.95)
ETA_DIS = 0.97                          # Eficiência de descarga eletroquímica
ETA_CH = 0.995                          # Eficiência coulômbica de recarga

# Potências de Consumo Incluindo OISL Contínuo (440 W)
P_AUX_KW = 0.56                         # Aviônica e bombeamento essencial [kW]
P_OISL_KW = 0.44                        # 4 terminais ópticos inter-satélite 24/7 [kW]
P_IT_ECL_KW = 18.0                      # Carga computacional prioritária no eclipse [kW]

# Cenários de Geração Solar Pós-Buck (nbuck = 0.91) e Carga de TI em Sol Sustentável
CENARIOS_FV = {                         # nome: (P_pv líquido [kW], P_IT sustentável em sol [kW])
    "EOL_85C": (46.14, 38.10),          # Ajustado para 38.10 kW devido ao OISL contínuo
    "EOL_65C": (49.73, 40.00),
    "EOL_28C": (54.18, 40.00),
}


def ocv(soc):
    """Relação linear unificada de circuito aberto: Voc = Ns * (3.0 + 1.1 * SoC)."""
    return NS * (3.0 + 1.1 * soc)


def r0_arrhenius(t_c):
    """Variação térmica da resistência interna de condução via lei de Arrhenius."""
    return R0_REF * math.exp(EA_K * (1.0 / (t_c + 273.15) - 1.0 / T_REF_K))


def simula_orbital(p_pv_kw, p_it_sol_kw, rth=RTH_COND, t_ch=T_CH, soc0=1.0,
                   t0=20.0, n_orb=10, dt=2.0, e_nom=E_NOM_J, cn=CN_AH):
    """
    Simula a dinâmica eletrotérmica multiorbital do BESS com carga de potência
    constante (CPL) no eclipse e recarga CC-CV no período iluminado.
    """
    p_aux_tot = P_AUX_KW + P_OISL_KW  # 1.00 kW total (plataforma + OISL)
    p_chg_w = max(0.0, p_pv_kw - (p_it_sol_kw + p_aux_tot) / ETA_DAB) * 1e3
    p_ecl_w = (P_IT_ECL_KW + p_aux_tot) * 1e3 / ETA_DAB  # 20.00 kW no barramento

    n = int(math.floor(n_orb * T_ORB / dt)) + 1
    soc, vt, tb, pj, ib = (np.zeros(n) for _ in range(5))
    ecl = np.zeros(n, dtype=bool)
    soc[0], tb[0] = soc0, t0
    a1 = math.exp(-dt / (RP1 * CP1))
    a2 = math.exp(-dt / (RP2 * CP2))
    vp1 = vp2 = 0.0

    for k in range(n - 1):
        ecl[k] = (k * dt) % T_ORB < T_ECL
        r0 = r0_arrhenius(tb[k])
        rs = r0 + R_DIST
        ve = ocv(soc[k]) - vp1 - vp2

        # Eclipse: Descarga como Carga de Potência Constante (Eq. 3.10)
        if ecl[k]:
            i = (ve - math.sqrt(max(ve * ve - 4.0 * rs * p_ecl_w, 0.0))) / (2.0 * rs)
            v = ve - i * rs
            dsoc = -(v * i * dt) / (ETA_DIS * e_nom)  # Formulação energética Eq. 3.7
        # Sol: Recarga controlada em dois estágios (CC -> CV)
        else:
            ich = (-ve + math.sqrt(ve * ve + 4.0 * rs * p_chg_w)) / (2.0 * rs)
            ich = min(ich, cn)  # Teto de 1,0C
            v = ve + ich * rs
            if v > V_OV:
                ich = max(0.0, (V_OV - ve) / rs)
                v = ve + ich * rs
            i = -ich
            dsoc = ETA_CH * (v * ich * dt) / e_nom

        soc[k + 1] = min(1.0, max(0.0, soc[k] + dsoc))
        vt[k], ib[k] = v, i
        vp1 = a1 * vp1 + RP1 * (1.0 - a1) * i
        vp2 = a2 * vp2 + RP2 * (1.0 - a2) * i
        pj[k] = i * i * r0 + vp1 ** 2 / RP1 + vp2 ** 2 / RP2

        # Controle ativo dos aquecedores de sobrevivência
        if tb[k] < T_ON:
            ph = P_HEATER
        elif tb[k] < T_OFF:
            ph = P_HEATER * (T_OFF - tb[k]) / (T_OFF - T_ON)
        else:
            ph = 0.0

        # Balanço térmico condutivo corrigido com o chassi estrutural
        tb[k + 1] = tb[k] + (pj[k] + ph - (tb[k] - t_ch) / rth) * dt / C_BAT

    ecl[-1], vt[-1], pj[-1], ib[-1] = ecl[-2], vt[-2], pj[-2], ib[-2]
    i0 = int((n_orb - 1) * T_ORB / dt)  # Amostragem da última órbita estabilizada
    sl = slice(i0, n - 1)
    m = ecl[sl]
    pj_med = float(np.mean(pj[sl]))

    resumo = dict(
        p_chg_kw=p_chg_w / 1e3,
        soc_min=float(soc[i0:].min()),
        v_min=float(vt[sl].min()),
        soc_fim=float(soc[-1]),
        i_ecl_med=float(ib[sl][m].mean()),
        i_ecl_pico=float(ib[sl][m].max()),
        pj_ecl=float(pj[sl][m].mean()),
        pj_sol=float(pj[sl][~m].mean()),
        pj_med=pj_med,
        tb_min=float(tb[i0:].min()),
        tb_pico=float(tb.max()),
        t_eq=t_ch + pj_med * rth,
        tau_h=C_BAT * rth / 3600.0
    )
    
    serie = dict(t=np.arange(n) * dt, soc=soc, vt=vt, tb=tb, pj=pj, ib=ib, ecl=ecl)
    return resumo, serie


def _psd(m, piso):
    """Simetriza a matriz e impõe piso positivo nos autovalores para garantir Cholesky."""
    s = 0.5 * (m + m.T)
    w, v = np.linalg.eigh(s)
    return v @ np.diag(np.maximum(w, piso)) @ v.T


def _perfil_i(t):
    """Perfil dinâmico com rajadas periódicas simulando cargas de tensores de IA."""
    return 25.0 + 15.0 * math.sin(2.0 * math.pi * t / 180.0) + (10.0 if (t % 120.0) < 30.0 else 0.0)


def executa_aukf(sigma_v, soc_real0=0.80, soc_est0=0.60, r0_real=R0_REF, cn_real=CN_AH,
                 i_bias=0.0, adapta_q=True, lim_r=(0.1, 10.0), piso=1e-8,
                 alpha=1e-3, beta=2.0, kappa=0.0, dt=1.0, n=1201, seed=42):
    """
    Executa o estimador AUKF de Sage-Husa completo (Q e R adaptativos) sob
    incertezas paramétricas da planta (R0 degradado, Cn reduzido e bias de sensor).
    """
    ruido = np.random.RandomState(seed).randn(n - 1)
    a1, a2 = math.exp(-dt / 57.6), math.exp(-dt / 528.0)
    s_t, v1, v2, v_cl, v_ms = (np.zeros(n) for _ in range(5))
    s_t[0] = soc_real0

    for k in range(n - 1):
        i = _perfil_i(k * dt)
        s_t[k + 1] = s_t[k] - i * dt / (cn_real * 3600.0)
        v1[k + 1] = a1 * v1[k] + RP1 * (1 - a1) * i
        v2[k + 1] = a2 * v2[k] + RP2 * (1 - a2) * i
        v_cl[k + 1] = ocv(s_t[k + 1]) - i * r0_real - v1[k + 1] - v2[k + 1]
        v_ms[k + 1] = v_cl[k + 1] + sigma_v * ruido[k]
    v_cl[0], v_ms[0] = v_cl[1], v_ms[1]

    L = 3
    lam = alpha ** 2 * (L + kappa) - L
    wm = np.full(2 * L + 1, 1.0 / (2.0 * (L + lam)))
    wc = wm.copy()
    wm[0] = lam / (L + lam)
    wc[0] = wm[0] + (1.0 - alpha ** 2 + beta)

    x = np.array([soc_est0, 0.0, 0.0])
    P = np.diag([0.05, 0.01, 0.01])
    Q = np.diag([1e-7, 1e-5, 1e-5])
    R, b = sigma_v ** 2, 0.95
    s_h, v_h = np.zeros(n), np.zeros(n)
    s_h[0] = x[0]

    for k in range(n - 1):
        i = _perfil_i(k * dt) + i_bias
        S = np.linalg.cholesky((L + lam) * _psd(P, piso))
        X = np.zeros((L, 2 * L + 1))
        X[:, 0] = x
        for j in range(L):
            X[:, 1 + j], X[:, 1 + L + j] = x + S[:, j], x - S[:, j]

        Xp = np.zeros_like(X)
        Xp[0] = X[0] - i * dt / (CN_AH * 3600.0)
        Xp[1] = a1 * X[1] + RP1 * (1 - a1) * i
        Xp[2] = a2 * X[2] + RP2 * (1 - a2) * i

        xp = Xp @ wm
        p_star = sum(wc[j] * np.outer(Xp[:, j] - xp, Xp[:, j] - xp) for j in range(2 * L + 1))
        Pp = p_star + Q

        Y = ocv(Xp[0]) - i * R0_REF - Xp[1] - Xp[2]
        yp = float(Y @ wm)
        pyy0 = sum(wc[j] * (Y[j] - yp) ** 2 for j in range(2 * L + 1))
        pxy = sum(wc[j] * (Xp[:, j] - xp) * (Y[j] - yp) for j in range(2 * L + 1))

        inov = v_ms[k + 1] - yp
        dk = (1.0 - b) / (1.0 - b ** (k + 1))
        R = min(lim_r[1] * sigma_v ** 2, max(lim_r[0] * sigma_v ** 2,
                (1.0 - dk) * R + dk * (inov ** 2 - pyy0)))

        pyy = pyy0 + R
        K = pxy / pyy
        x = xp + K * inov
        P = Pp - np.outer(K, K) * pyy

        if adapta_q:
            Q = _psd((1 - dk) * Q + dk * (inov ** 2 * np.outer(K, K) + P - p_star), 1e-12)

        x[0] = min(1.0, max(0.0, x[0]))
        s_h[k + 1] = x[0]
        v_h[k + 1] = ocv(x[0]) - i * R0_REF - x[1] - x[2]

    err = np.abs(s_h - s_t) * 100.0
    idx = np.where(err > 1.0)[0]
    
    resumo = dict(
        t_conv=(int(idx[-1]) + 1) * dt if len(idx) else 0.0,
        err_med=float(err[59:].mean()),
        err_max=float(err[59:].max()),
        rmse_mV=float(1e3 * np.sqrt(np.mean((v_h[59:] - v_cl[59:]) ** 2)))
    )
    serie = dict(t=np.arange(n) * dt, s_t=s_t, s_h=s_h, err=err, v_cl=v_cl, v_h=v_h)
    return resumo, serie

import math

# ---- primário GS66516T (8 chaves, 2 por braço)
RON25, ALPHA_PRI = 25e-3, 0.0128                 # [DS] / [VERIFICAR curva Ron x Tj]
RTH_JC, RTH_TIM = 0.27, 0.85                     # [DS] / [HIP]
RTH = RTH_JC + RTH_TIM
I_IN, D_EFF, N_PAR_PRI = 51.2, 0.45, 2
V_IN, FSW = 400.0, 150e3
TR, TF, ISW_ON = 12e-9, 15e-9, 25.0              # [HIP]
COSS, QG, VGATE = 207e-12, 14.2e-9, 6.0          # [VERIFICAR] Qg (12,1 nC máx em listagem comercial)
IRMS = I_IN * math.sqrt(D_EFF)                   # 34,35 A por chave (arranjo de 4 chaves)

# ---- secundário IG1NT052N10R
R_SEC25, ALPHA_SEC, I_D_MAX = 6e-3, 0.004, 52.0  # [DS-web] / [EST] / [DS-web]
N_PAR_SEC, R_INTERLIG, I_OUT = 12, 0.732e-3, 416.7   # derating adotado / [HIP]

# ---- magnéticos
K_FE, A_F, B_F, V_E, DB = 0.70, 1.46, 2.52, 4.5e-4, 0.15
RAC_PRI, RAC_SEC = 15e-3, 0.25e-3

# ---- capacitores 48 V
C_NOM_UNIT, FRAC_BIAS, DF_MAX = 22e-6, 0.50, 0.025   # [DS] / [EST] / [DS]
C_BANCO_BOL, FATOR_EOL = 15e-3, 0.70
ESR_BOL, ESR_EOL = 1.2e-3, 2.4e-3                    # [HIP]
LIM_48 = 11.52                                       # mΩ

Perdas = namedtuple("Perdas", "p_cp p_sw p_cs_disp p_cs_int p_core p_cu total q_primario eta")


def ron_pri(tj):
    return RON25 * (1.0 + ALPHA_PRI * (tj - 25.0))


def r_sec(tj=80.0):
    return R_SEC25 * (1.0 + ALPHA_SEC * (tj - 25.0))


def p_sw_dev(n):
    p_ov = V_IN * ISW_ON / n * FSW * (TR + TF) / 2.0
    return p_ov + 0.5 * COSS * V_IN ** 2 * FSW + QG * VGATE * FSW


def tj_eq(tb, n=N_PAR_PRI):
    a = (IRMS / n) ** 2 * RON25
    c0 = a * (1.0 - ALPHA_PRI * 25.0) + p_sw_dev(n)
    return (tb + RTH * c0) / (1.0 - RTH * a * ALPHA_PRI)


def tb_max(tj, n=N_PAR_PRI):
    a = (IRMS / n) ** 2 * RON25
    c0 = a * (1.0 - ALPHA_PRI * 25.0) + p_sw_dev(n)
    return tj * (1.0 - RTH * a * ALPHA_PRI) - RTH * c0


def perdas(n_pri=N_PAR_PRI, n_sec=N_PAR_SEC, r_int=R_INTERLIG):
    nsw = 4 * n_pri
    r_disp = 2.0 * r_sec() / n_sec
    p_cp = nsw * (IRMS / n_pri) ** 2 * ron_pri(80.0)
    p_sw = nsw * p_sw_dev(n_pri)
    p_cs_disp = I_OUT ** 2 * r_disp
    p_cs_int = I_OUT ** 2 * r_int
    p_core = K_FE * FSW ** A_F * DB ** B_F * V_E
    p_cu = 2.0 * IRMS ** 2 * RAC_PRI + I_OUT ** 2 * RAC_SEC   # enrolamento = sqrt(2) * I_chave
    tot = p_cp + p_sw + p_cs_disp + p_cs_int + p_core + p_cu
    eta = 100.0 * 20000.0 / (20000.0 + tot)
    return Perdas(p_cp, p_sw, p_cs_disp, p_cs_int, p_core, p_cu, tot, p_cp + p_sw, eta)


def zout_pico(L, R, C, ESR, n=8001):
    f = np.logspace(1, 5.3, n)
    s = 2j * np.pi * f
    z = np.abs((s * L + R) * (s * C * ESR + 1.0) / (s * s * L * C + s * C * (R + ESR) + 1.0))
    k = int(np.argmax(z))
    return z[k] * 1e3, f[k]


def zout_curva(L, R, C, ESR, n=8001):
    f = np.logspace(1, 5.3, n)
    s = 2j * np.pi * f
    z = np.abs((s * L + R) * (s * C * ESR + 1.0) / (s * s * L * C + s * C * (R + ESR) + 1.0))
    return f, z * 1e3


def regioes_ok(L, R, esr, limite, c_min=1.0e-3, c_max=15.0e-3, n=141):
    cs = np.linspace(c_min, c_max, n)
    ok = [zout_pico(L, R, c, esr)[0] <= limite for c in cs]
    regs, ini = [], None
    for i in range(len(ok)):
        if ok[i] and ini is None:
            ini = i
        if (not ok[i]) and ini is not None:
            regs.append((cs[ini], cs[i - 1]))
            ini = None
    if ini is not None:
        regs.append((cs[ini], cs[-1]))
    return regs


def degrau_cpl(c, esr, vs=400.0, L=1.8e-6, R=14e-3, p1=2000.0, p2=20000.0, dt=5e-7, t_fim=0.01):
    vb = vs
    for _ in range(60):
        vb = vs - R * p1 / vb
    i, vc, v0, vmin = p1 / vb, vb, vb, 1e9
    
    t_arr = []
    v_arr = []
    for k in range(int(t_fim / dt)):
        t = k * dt
        p = p1 if t < 0.5e-3 else p2
        vbb = vc
        for _ in range(8):
            vbb = vc + esr * (i - p / vbb)
        il = p / vbb
        i += (vs - vbb - R * i) / L * dt
        vc += (i - il) / c * dt
        vmin = min(vmin, vbb)
        t_arr.append(t)
        v_arr.append(vbb)
    return v0 - vmin, t_arr, v_arr

def clamp(value, lower, upper):
    return max(lower, min(upper, value))


def solar_declination_approx(date):
    day = date.timetuple().tm_yday
    hour = date.hour + date.minute / 60.0 + date.second / 3600.0
    fractional_day = day + (hour - 12.0) / 24.0
    gamma = math.radians(
        360.0 * (fractional_day - 81.0) / 365.0
    )
    return 23.44 * math.sin(gamma)


def beta_from_ltan(
    date,
    ltan_hours,
    inclination_deg=SSO_INCLINATION_DEG,
):
    if not 0.0 <= ltan_hours < 24.0:
        raise ValueError("LTAN deve estar em [0,24).")

    delta = math.radians(solar_declination_approx(date))
    inclination = math.radians(inclination_deg)
    delta_omega = math.radians(15.0 * (ltan_hours - 12.0))

    sin_beta = (
        math.sin(inclination)
        * math.cos(delta)
        * math.sin(delta_omega)
        + math.cos(inclination)
        * math.sin(delta)
    )

    return math.degrees(
        math.asin(clamp(sin_beta, -1.0, 1.0))
    )


def beta_for_orbit(beta_profile_deg, orbit_index):
    if isinstance(beta_profile_deg, (int, float)):
        return float(beta_profile_deg)

    if beta_profile_deg is None or len(beta_profile_deg) == 0:
        raise ValueError(
            "beta_profile_deg deve ser escalar ou sequência não vazia."
        )

    return float(
        beta_profile_deg[
            min(orbit_index, len(beta_profile_deg) - 1)
        ]
    )

# 4. Geometria de eclipse
# ==============================================================================

def circle_overlap_area(radius_sun, radius_earth, separation):
    if separation >= radius_sun + radius_earth:
        return 0.0

    if separation <= abs(radius_earth - radius_sun):
        return math.pi * min(radius_sun, radius_earth) ** 2

    arg_sun = clamp(
        (
            separation**2
            + radius_sun**2
            - radius_earth**2
        ) / (2.0 * separation * radius_sun),
        -1.0,
        1.0,
    )

    arg_earth = clamp(
        (
            separation**2
            + radius_earth**2
            - radius_sun**2
        ) / (2.0 * separation * radius_earth),
        -1.0,
        1.0,
    )

    radicand = max(
        0.0,
        (-separation + radius_sun + radius_earth)
        * (separation + radius_sun - radius_earth)
        * (separation - radius_earth + radius_sun)
        * (separation + radius_earth + radius_sun),
    )

    return (
        radius_sun**2 * math.acos(arg_sun)
        + radius_earth**2 * math.acos(arg_earth)
        - 0.5 * math.sqrt(radicand)
    )


def solar_disk_fraction(theta_rad, beta_deg, altitude_km):
    radius_orbit_km = RE_KM + altitude_km

    alpha_earth = math.asin(
        clamp(RE_KM / radius_orbit_km, -1.0, 1.0)
    )
    alpha_sun = math.asin(
        clamp(RSUN_KM / D_ES_KM, -1.0, 1.0)
    )

    beta = math.radians(beta_deg)
    cos_gamma = -math.cos(beta) * math.cos(theta_rad)
    gamma = math.acos(clamp(cos_gamma, -1.0, 1.0))

    overlap = circle_overlap_area(
        alpha_sun,
        alpha_earth,
        gamma,
    )

    fraction = 1.0 - overlap / (math.pi * alpha_sun**2)
    return clamp(fraction, 0.0, 1.0)

# 5. Fator de visão e ambiente radiativo
# ==============================================================================

def earth_view_factor(alt_km, phi_deg):
    H = (RE_KM + alt_km) / RE_KM

    if H <= 1.0:
        raise ValueError("A altitude deve ser positiva.")

    phi = math.radians(phi_deg)
    theta_m = math.asin(1.0 / H)
    root = math.sqrt(H**2 - 1.0)

    if phi <= math.pi / 2.0 - theta_m:
        return clamp(math.cos(phi) / H**2, 0.0, 1.0)

    if phi <= math.pi / 2.0 + theta_m:
        sin_phi = math.sin(phi)
        if abs(sin_phi) < 1e-12:
            return 0.0

        arg1 = clamp(root / (H * sin_phi), -1.0, 1.0)
        arg2 = clamp(
            root * math.cos(phi) / sin_phi,
            -1.0,
            1.0,
        )
        radicand = max(
            0.0,
            1.0 - H**2 * math.cos(phi)**2,
        )

        factor = (
            0.5
            - math.asin(arg1) / math.pi
            + math.cos(phi)
            / (math.pi * H**2)
            * math.acos(arg2)
            - root
            * math.sqrt(radicand)
            / (math.pi * H**2)
        )
        return clamp(factor, 0.0, 1.0)

    return 0.0


def orbital_geometry(theta_rad, beta_deg):
    beta = math.radians(beta_deg)

    radial_hat = (
        math.cos(theta_rad),
        math.sin(theta_rad),
        0.0,
    )

    sun_hat = (
        math.cos(beta),
        0.0,
        math.sin(beta),
    )

    nadir_hat = tuple(-value for value in radial_hat)
    radiator_hat = tuple(-value for value in sun_hat)

    cos_phi = sum(
        radiator_hat[i] * nadir_hat[i]
        for i in range(3)
    )
    phi_deg = math.degrees(
        math.acos(clamp(cos_phi, -1.0, 1.0))
    )

    mu_albedo = max(
        0.0,
        sum(radial_hat[i] * sun_hat[i] for i in range(3)),
    )

    mu_solar_nominal = max(
        0.0,
        sum(radiator_hat[i] * sun_hat[i] for i in range(3)),
    )

    return phi_deg, mu_albedo, mu_solar_nominal


def thermal_environment(
    theta_rad,
    beta_deg,
    altitude_km,
    albedo=ALBEDO,
    q_earth_ir_wm2=Q_EARTH_IR,
    solar_fraction=1.0,
    attitude_error_deg=0.0,
):
    phi_deg, mu_albedo, mu_solar_nominal = orbital_geometry(
        theta_rad,
        beta_deg,
    )

    f_earth = earth_view_factor(
        altitude_km,
        phi_deg,
    )

    mu_solar_error = max(
        0.0,
        math.sin(math.radians(abs(attitude_error_deg))),
    )
    mu_solar = max(mu_solar_nominal, mu_solar_error)

    q_solar_abs = (
        ALPHA_RAD
        * G_SOL
        * mu_solar
        * solar_fraction
    )

    q_albedo_abs = (
        ALPHA_RAD
        * G_SOL
        * albedo
        * f_earth
        * mu_albedo
    )

    q_ir_abs = EPS_RAD * q_earth_ir_wm2 * f_earth

    return {
        "f_earth": f_earth,
        "phi_deg": phi_deg,
        "mu_albedo": mu_albedo,
        "mu_solar": mu_solar,
        "q_solar_abs": q_solar_abs,
        "q_albedo_abs": q_albedo_abs,
        "q_ir_abs": q_ir_abs,
        "q_ambient_abs": (
            q_solar_abs
            + q_albedo_abs
            + q_ir_abs
        ),
    }

# 6. Modelo energético da bateria

def battery_loss(p_terminal_w):
    current_a = abs(p_terminal_w) / V_BAT
    return R_BAT * current_a**2


def max_discharge_for_energy(
    p_requested_w,
    energy_available_j,
    dt_s,
):
    if (
        p_requested_w <= 0.0
        or energy_available_j <= 0.0
        or dt_s <= 0.0
    ):
        return 0.0, 0.0

    def chemical_energy_required(power_w):
        return (power_w + battery_loss(power_w)) * dt_s

    if chemical_energy_required(p_requested_w) <= energy_available_j:
        return p_requested_w, battery_loss(p_requested_w)

    low = 0.0
    high = p_requested_w

    for _ in range(60):
        middle = 0.5 * (low + high)
        if chemical_energy_required(middle) <= energy_available_j:
            low = middle
        else:
            high = middle

    return low, battery_loss(low)


def max_charge_for_headroom(
    p_available_w,
    energy_headroom_j,
    dt_s,
):
    if (
        p_available_w <= 0.0
        or energy_headroom_j <= 0.0
        or dt_s <= 0.0
    ):
        return 0.0, 0.0, 0.0

    p_limit = min(p_available_w, P_MAX_CHARGE_W)

    def chemical_energy_stored(power_w):
        return max(
            0.0,
            power_w - battery_loss(power_w),
        ) * dt_s

    if chemical_energy_stored(p_limit) <= energy_headroom_j:
        loss_w = battery_loss(p_limit)
        return (
            p_limit,
            loss_w,
            max(0.0, p_limit - loss_w),
        )

    low = 0.0
    high = p_limit

    for _ in range(60):
        middle = 0.5 * (low + high)
        if chemical_energy_stored(middle) <= energy_headroom_j:
            low = middle
        else:
            high = middle

    loss_w = battery_loss(low)
    return (
        low,
        loss_w,
        max(0.0, low - loss_w),
    )

def simulate(

    albedo=ALBEDO,

    heater_set_c=HEATER_SET_C,
    heater_kp=HEATER_KP,
    heater_max_w=HEATER_MAX_W,
    attitude_error_deg=0.0,
    eta_dab=ETA_DAB,
    n_orbits_max=MAX_ORBITS,
    start_date=START_DATE,  # <--- Adicionado aqui

):
    if beta_profile_deg is None:
        raise ValueError(
            "Informe beta_profile_deg: escalar ou sequência por órbita."

    if min(ti_sun_w, ti_eclipse_w, pv_input_kw, heater_max_w) < 0.0:
        raise ValueError("Potências não podem ser negativas.")

    if heater_kp < 0.0 or not 0.0 < eta_dab <= 1.0:
        raise ValueError("heater_kp ou eta_dab inválido.")

    if not (0.0 <= SOC_MIN < SOC_MAX <= 1.0):
        raise ValueError("Limites de SoC inválidos.")

    cap_j = E_BAT_KWH * 3.6e6
    c_rad = A_RAD * RHO_AREAL * CP_RAD
    n_steps = max(1, int(round(T_ORB / dt_s)))
    dt = T_ORB / n_steps

    tr = 273.15 + T_RAD_INIT_C
    tc = 273.15 + T_CHASSIS_INIT_C
    tb = 273.15 + T_BAT_INIT_C
    soc = SOC_MAX

    previous_start = None
    previous_envelope = None
    stable_state_count = 0
    stable_envelope_count = 0

    global_soc_min = soc
    final_result = None

    profile_is_constant = isinstance(
        beta_profile_deg,
        (int, float),
    )

    if profile_is_constant:
        beta_description = (
            f"beta constante={float(beta_profile_deg):.4f} deg"
        )
    else:
        beta_description = (
            f"perfil beta com {len(beta_profile_deg)} valores"
        )

    history_records = []
    elapsed_total_s = 0.0

    for orbit in range(1, n_orbits_max + 1):
        beta_now = beta_for_orbit(
            beta_profile_deg,
            orbit - 1,
        )

        start_state = (tr, tc, tb, soc)
        thermal_energy_start = (
            c_rad * tr
            + C_CHASSIS * tc
            + C_BAT * tb
        )
        soc_start = soc

        tr_min = tc_min = tb_min = float("inf")
        tr_max = tc_max = tb_max = float("-inf")
        soc_min_orbit = soc

        e_pv_available = 0.0
        e_pv_used = 0.0
        e_pv_curtailed = 0.0
        e_load_served = 0.0
        e_bat_charge_terminal = 0.0
        e_bat_discharge_terminal = 0.0
        e_bat_loss = 0.0
        e_heater = 0.0
        e_dab_loss = 0.0
        e_ti_heat = 0.0
        e_base_heat = 0.0
        e_chassis_heat = 0.0
        e_radiated = 0.0
        e_ambient = 0.0
        e_unmet = 0.0

        for k in range(n_steps):
            theta = 2.0 * math.pi * (k * dt) / T_ORB

            solar_fraction = solar_disk_fraction(
                theta,
                beta_now,
                ALTITUDE_KM,
            )

            p_ti_requested = (
                ti_eclipse_w
                + (ti_sun_w - ti_eclipse_w) * solar_fraction
            )

            p_pv_available = (
                pv_input_kw * 1000.0 * solar_fraction
            )

            env = thermal_environment(
                theta_rad=theta,
                beta_deg=beta_now,
                altitude_km=ALTITUDE_KM,
                albedo=albedo,
                q_earth_ir_wm2=q_earth_ir_wm2,
                solar_fraction=solar_fraction,
                attitude_error_deg=attitude_error_deg,
            )

            tb_c = tb - 273.15
            p_heater_requested = min(
                heater_max_w,
                max(
                    0.0,
                    heater_kp * (heater_set_c - tb_c),
                ),
            )

            p_ti_bus_requested = p_ti_requested / eta_dab
            p_total_requested = (
                p_ti_bus_requested
                + P_BASE
                + p_heater_requested
            )

            p_pv_to_load = min(
                p_pv_available,
                p_total_requested,
            )

            p_surplus = max(
                0.0,
                p_pv_available - p_pv_to_load,
            )

            p_deficit = max(
                0.0,
                p_total_requested - p_pv_to_load,
            )

            p_bat_charge_terminal = 0.0
            p_bat_discharge_terminal = 0.0
            p_bat_loss_charge = 0.0
            p_bat_loss_discharge = 0.0
            p_bat_chemical_charge = 0.0

            if p_deficit > 0.0:
                energy_available = max(
                    0.0,
                    (soc - SOC_MIN) * cap_j,
                )

                (
                    p_bat_discharge_terminal,
                    p_bat_loss_discharge,
                ) = max_discharge_for_energy(
                    p_deficit,
                    energy_available,
                    dt,
                )

                chemical_draw = (
                    p_bat_discharge_terminal
                    + p_bat_loss_discharge
                ) * dt

                soc = max(
                    SOC_MIN,
                    soc - chemical_draw / cap_j,
                )

            elif p_surplus > 0.0:
                energy_headroom = max(
                    0.0,
                    (SOC_MAX - soc) * cap_j,
                )

                (
                    p_bat_charge_terminal,
                    p_bat_loss_charge,
                    p_bat_chemical_charge,
                ) = max_charge_for_headroom(
                    p_surplus,
                    energy_headroom,
                    dt,
                )

                soc = min(
                    SOC_MAX,
                    soc
                    + p_bat_chemical_charge * dt / cap_j,
                )

            p_bat_loss = (
                p_bat_loss_charge
                + p_bat_loss_discharge
            )

            p_load_served = (
                p_pv_to_load
                + p_bat_discharge_terminal
            )

            p_unmet = max(
                0.0,
                p_total_requested - p_load_served,
            )

            e_unmet += p_unmet * dt

            p_pv_used = (
                p_pv_to_load
                + p_bat_charge_terminal
            )

            p_pv_curtailed = max(
                0.0,
                p_pv_available - p_pv_used,
            )

            served_fraction = clamp(
                p_load_served / max(p_total_requested, 1e-12),
                0.0,
                1.0,
            )

            p_ti_served = p_ti_requested * served_fraction
            p_heater_served = p_heater_requested * served_fraction

            p_ti_bus_served = p_ti_served / eta_dab
            p_dab_loss = max(
                0.0,
                p_ti_bus_served - p_ti_served,
            )

            q_ti = p_ti_served
            q_base = Q_BASE_THERMAL * served_fraction
            q_dab = p_dab_loss

            q_chassis_source = q_ti + q_base + q_dab

            q_bat_to_chassis = (tb - tc) / r_bat_ch
            q_chassis_to_rad = (tc - tr) / r_ch_rad

            q_ambient = A_RAD * env["q_ambient_abs"]

            q_radiated = (
                A_RAD
                * ETA_FIN
                * EPS_RAD
                * SIGMA
                * (max(tr, 1.0) ** 4 - 3.0**4)
            )

            d_tb = (
                p_bat_loss
                + p_heater_served
                - q_bat_to_chassis
            ) / C_BAT

            d_tc = (
                q_chassis_source
                + q_bat_to_chassis
                - q_chassis_to_rad
            ) / C_CHASSIS

            d_tr = (
                q_chassis_to_rad
                + q_ambient
                - q_radiated
            ) / c_rad

            tb += d_tb * dt
            tc += d_tc * dt
            tr += d_tr * dt

            e_pv_available += p_pv_available * dt
            e_pv_used += p_pv_used * dt
            e_pv_curtailed += p_pv_curtailed * dt
            e_load_served += p_load_served * dt

            e_bat_charge_terminal += (
                p_bat_charge_terminal * dt
            )
            e_bat_discharge_terminal += (
                p_bat_discharge_terminal * dt
            )
            e_bat_loss += p_bat_loss * dt
            e_heater += p_heater_served * dt

            e_dab_loss += p_dab_loss * dt
            e_ti_heat += q_ti * dt
            e_base_heat += q_base * dt
            e_chassis_heat += q_chassis_source * dt

            e_radiated += q_radiated * dt
            e_ambient += q_ambient * dt

            tr_c = tr - 273.15
            tc_c = tc - 273.15
            tb_c_new = tb - 273.15

            tr_min = min(tr_min, tr_c)
            tr_max = max(tr_max, tr_c)
            tc_min = min(tc_min, tc_c)
            tc_max = max(tc_max, tc_c)
            tb_min = min(tb_min, tb_c_new)
            tb_max = max(tb_max, tb_c_new)

            soc_min_orbit = min(soc_min_orbit, soc)
            global_soc_min = min(
                global_soc_min if "global_soc_min" in locals() else SOC_MAX,
                soc,
            )

            if record_history:
                history_records.append({
                    "orbit": orbit,
                    "time_s": elapsed_total_s,
                    "t_rad_c": tr_c,
                    "t_chassis_c": tc_c,
                    "t_bat_c": tb_c_new,
                    "soc": soc,
                    "solar_fraction": solar_fraction,
                })

            elapsed_total_s += dt

        thermal_energy_end = (
            c_rad * tr
            + C_CHASSIS * tc
            + C_BAT * tb
        )

        delta_thermal = (
            thermal_energy_end - thermal_energy_start
        )

        electrical_residual = (
            e_pv_used
            + e_bat_discharge_terminal
            - e_load_served
            - e_bat_charge_terminal
        )

        battery_chemical_change = (
            (soc - soc_start) * cap_j
        )

        battery_energy_residual = (
            battery_chemical_change
            - (
                e_bat_charge_terminal
                - e_bat_discharge_terminal
                - e_bat_loss
            )
        )

        thermal_residual = (
            e_chassis_heat
            + e_bat_loss
            + e_heater
            + e_ambient
            - e_radiated
            - delta_thermal
        )

        end_state = (tr, tc, tb, soc)
        envelope = (
            tr_min,
            tr_max,
            tc_min,
            tc_max,
            tb_min,
            tb_max,
        )

        if previous_start is None:
            delta_t_state = float("inf")
            delta_soc_state = float("inf")
            delta_envelope = float("inf")
        else:
            delta_t_state = max(
                abs(end_state[i] - previous_start[i])
                for i in range(3)
            )
            delta_soc_state = abs(
                end_state[3] - previous_start[3]
            )
            delta_envelope = max(
                abs(envelope[i] - previous_envelope[i])
                for i in range(6)
            )

        periodic_candidate = (
            isinstance(beta_profile_deg, (int, float))
            and delta_t_state < TOL_T_K
            and delta_soc_state < TOL_SOC
        )

        operationally_feasible = (
            e_unmet < 1.0
            and soc > SOC_MIN + TOL_SOC
        )

        if periodic_candidate and operationally_feasible:
            stable_state_count += 1
        else:
            stable_state_count = 0

        envelope_candidate = (
            delta_envelope < TOL_ENVELOPE_K
        )

        if envelope_candidate and operationally_feasible:
            stable_envelope_count += 1
        else:
            stable_envelope_count = 0

        periodic_state_converged = (
            stable_state_count >= 2
        )
        envelope_converged = (
            stable_envelope_count >= 2
        )
        converged = (
            operationally_feasible
            and (
                periodic_state_converged
                or envelope_converged
            )
        )

        final_result = Result(
            converged=converged,
            envelope_converged=envelope_converged,
            periodic_state_converged=periodic_state_converged,
            operationally_feasible=operationally_feasible,
            orbits=orbit,
            beta_description=beta_description,

            t_rad_min_c=tr_min,
            t_rad_max_c=tr_max,
            t_ch_min_c=tc_min,
            t_ch_max_c=tc_max,
            t_bat_min_c=tb_min,
            t_bat_max_c=tb_max,

            soc_min_last_orbit=soc_min_orbit,
            soc_min_run=global_soc_min,
            soc_final=soc,
            unmet_energy_j_last_orbit=e_unmet,

            e_pv_available_j_last_orbit=e_pv_available,
            e_pv_used_j_last_orbit=e_pv_used,
            e_pv_curtailed_j_last_orbit=e_pv_curtailed,
            e_load_served_j_last_orbit=e_load_served,

            e_bat_charge_terminal_j_last_orbit=e_bat_charge_terminal,
            e_bat_discharge_terminal_j_last_orbit=e_bat_discharge_terminal,
            e_bat_chemical_change_j_last_orbit=battery_chemical_change,
            e_battery_loss_j_last_orbit=e_bat_loss,
            e_heater_j_last_orbit=e_heater,

            e_dab_loss_j_last_orbit=e_dab_loss,
            e_ti_heat_j_last_orbit=e_ti_heat,
            e_base_heat_j_last_orbit=e_base_heat,
            e_chassis_heat_j_last_orbit=e_chassis_heat,

            e_radiated_j_last_orbit=e_radiated,
            e_ambient_j_last_orbit=e_ambient,

            electrical_residual_j_last_orbit=electrical_residual,
            battery_energy_residual_j_last_orbit=battery_energy_residual,
            thermal_residual_j_last_orbit=thermal_residual,

            delta_t_state_k=delta_t_state,
            delta_soc_state=delta_soc_state,
            delta_envelope_k=delta_envelope,
        )

        if converged:
            break

        previous_start = start_state
        previous_envelope = envelope

    if record_history:
        return final_result, history_records
    return final_result

# 9. Diagnósticos de aleta e temperatura quase-estática de junção
# ==============================================================================

def fin_efficiency_diagnostic():
    t_ref_k = 318.15
    h_rad = 4.0 * EPS_RAD * SIGMA * t_ref_k**3
    k_material = 1275.0  # W/(m K), hipótese
    thickness = 0.002    # m, hipótese
    length = 0.075       # m, hipótese

    m = math.sqrt(
        2.0 * h_rad / (k_material * thickness)
    )
    ml = m * length
    eta_fin_calculated = (
        math.tanh(ml) / ml if ml > 0.0 else 1.0
    )
    return h_rad, m, eta_fin_calculated

def rank_ltan_by_insolation(
    start_date,
    altitude_km=ALTITUDE_KM,
    inclination_deg=SSO_INCLINATION_DEG,
    ltan_step_hours=0.5,
    day_step=5,
    n_phase_samples=120,
):
    if ltan_step_hours <= 0.0:
        raise ValueError("ltan_step_hours deve ser positivo.")
    if day_step <= 0 or n_phase_samples <= 0:
        raise ValueError("day_step e n_phase_samples devem ser positivos.")

    n_ltan = int(round(24.0 / ltan_step_hours))
    candidates = []
    worst_beta_dates = {}

    for j in range(n_ltan):
        ltan = j * ltan_step_hours
        total_fraction = 0.0
        count = 0

        for day in range(0, 365, day_step):
            date = start_date + timedelta(days=day)
            beta = beta_from_ltan(
                date,
                ltan,
                inclination_deg,
            )

            for k in range(n_phase_samples):
                theta = (
                    2.0 * math.pi * k / n_phase_samples
                )
                total_fraction += solar_disk_fraction(
                    theta,
                    beta,
                    altitude_km,
                )
                count += 1

        candidates.append(
            (ltan, total_fraction / count)
        )

        daily_betas = []
        for day in range(365):
            date = start_date + timedelta(days=day)
            beta = beta_from_ltan(
                date,
                ltan,
                inclination_deg,
            )
            daily_betas.append((abs(beta), date))

        worst_beta_dates[ltan] = min(
            daily_betas,
            key=lambda item: item[0],
        )[1]

    worst = min(candidates, key=lambda item: item[1])
    best = max(candidates, key=lambda item: item[1])

    return {
        "worst_ltan_h": worst[0],
        "worst_mean_solar_fraction": worst[1],
        "best_ltan_h": best[0],
        "best_mean_solar_fraction": best[1],
        "worst_beta_dates": worst_beta_dates,
        "ranking": sorted(candidates, key=lambda item: item[1]),
    }


def build_beta_profile(
    start_date,
    ltan_hours,
    n_orbits,
    orbit_period_s_value=T_ORB,
    inclination_deg=SSO_INCLINATION_DEG,
):
    return [
        beta_from_ltan(
            start_date + timedelta(
                seconds=i * orbit_period_s_value
            ),
            ltan_hours,
            inclination_deg,
        )
        for i in range(n_orbits)
    ]

def print_result(label, ltan, beta_profile, result):
    print(f"\n--- {label} ---")
    print(f"LTAN = {ltan:05.2f} h")
    print(
        "Beta do perfil (min/max) = "
        f"{min(beta_profile):.3f} / {max(beta_profile):.3f} deg"
    )
    print(f"Órbitas simuladas = {result.orbits}")
    print(
        "Convergência geral / envelope / estado periódico / "
        "viabilidade = "
        f"{result.converged} / "
        f"{result.envelope_converged} / "
        f"{result.periodic_state_converged} / "
        f"{result.operationally_feasible}"
    )

    print(
        f"Radiador min/max = "
        f"{result.t_rad_min_c:.2f} / {result.t_rad_max_c:.2f} °C"
    )
    print(
        f"Chassi min/max = "
        f"{result.t_ch_min_c:.2f} / {result.t_ch_max_c:.2f} °C"
    )
    print(
        f"Bateria min/max = "
        f"{result.t_bat_min_c:.2f} / {result.t_bat_max_c:.2f} °C"
    )

    print(
        f"SoC mínimo última órbita/global = "
        f"{100.0 * result.soc_min_last_orbit:.2f} / "
        f"{100.0 * result.soc_min_run:.2f} %"
    )
    print(
        f"SoC final = {100.0 * result.soc_final:.2f} %"
    )

    print(
        "Energia não atendida última órbita = "
        f"{result.unmet_energy_j_last_orbit / 3.6e6:.6f} kWh"
    )

def save_outputs(case_label, history_records, result):
    output_dir = OUT_DIR
    os.makedirs(output_dir, exist_ok=True)
    
    safe_label = case_label.lower().replace(" ", "_")
    csv_filename = os.path.join(output_dir, f"simulacao_{safe_label}.csv")

    # 1. Salvar CSV com a série temporal
    if history_records:
        keys = history_records[0].keys()
        with open(csv_filename, mode="w", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=keys)
            writer.writeheader()
            for row in history_records:
                writer.writerow(row)
        print(f"-> Arquivo CSV salvo com sucesso em: {csv_filename}")

    # 2. Gerar Gráficos PNG (caso matplotlib esteja instalado)
    if HAS_MATPLOTLIB and history_records:
        times_h = [r["time_s"] / 3600.0 for r in history_records]
        t_rad = [r["t_rad_c"] for r in history_records]
        t_ch = [r["t_chassis_c"] for r in history_records]
        t_bat = [r["t_bat_c"] for r in history_records]
        socs = [100.0 * r["soc"] for r in history_records]

        # Gráfico de Temperaturas
        plt.figure(figsize=(10, 5))
        plt.plot(times_h, t_rad, label="Radiador (°C)", color="tab:blue")
        plt.plot(times_h, t_ch, label="Chassi (°C)", color="tab:orange")
        plt.plot(times_h, t_bat, label="Bateria (°C)", color="tab:green")
        plt.title(f"Evolução Térmica — {case_label}")
        plt.xlabel("Tempo de Simulação (horas)")
        plt.ylabel("Temperatura (°C)")
        plt.grid(True, linestyle="--", alpha=0.6)
        plt.legend()
        plt.tight_layout()
        plot_temp_path = os.path.join(output_dir, f"temperaturas_{safe_label}.png")
        plt.savefig(plot_temp_path, dpi=300)
        plt.close()
        print(f"-> Gráfico de temperaturas salvo em: {plot_temp_path}")

        # Gráfico de Estado de Carga (SoC)
        plt.figure(figsize=(10, 4))
        plt.plot(times_h, socs, label="SoC (%)", color="tab:purple")
        plt.axhline(y=SOC_MIN * 100, color="r", linestyle=":", label=f"SoC Min ({SOC_MIN*100}%)")
        plt.title(f"Estado de Carga da Bateria (SoC) — {case_label}")
        plt.xlabel("Tempo de Simulação (horas)")
        plt.ylabel("SoC (%)")
        plt.grid(True, linestyle="--", alpha=0.6)
        plt.legend()
        plt.tight_layout()
        plot_soc_path = os.path.join(output_dir, f"soc_{safe_label}.png")
        plt.savefig(plot_soc_path, dpi=300)
        plt.close()
        print(f"-> Gráfico de SoC salvo em: {plot_soc_path}")
    elif not HAS_MATPLOTLIB:
        print("-> Matplotlib não encontrado. Apenas os dados CSV foram salvos.")

def main():
    print("SIMULAÇÃO 4.1 — extremos aproximados de LTAN/insolação")
    print(
        f"Altitude={ALTITUDE_KM:.1f} km; "
        f"período={T_ORB:.2f} s; "
        f"inclinação provisória={SSO_INCLINATION_DEG:.2f}°"
    )
    print(
        f"Data inicial={START_DATE.isoformat()}; "
        f"passo={SIM_DT_S:g} s; "
        f"PV pós-Buck={PV_SCENARIOS_KW[PV_CASE]:.2f} kW"
    )
    print(f"Área emissora total fixa: {A_RAD:.1f} m²")

    h_fin, m_fin, eta_fin_calc = fin_efficiency_diagnostic()
    print(
        "Diagnóstico separado de aleta: "
        f"h_rad={h_fin:.3f} W/(m² K); "
        f"m={m_fin:.3f} 1/m; "
        f"eta calculada={eta_fin_calc:.5f}; "
        f"eta adotada={ETA_FIN:.3f}"
    )

    ranking = rank_ltan_by_insolation(START_DATE)

    print("\nVarredura anual aproximada de LTAN:")
    print(
        f"Menor fração solar média: LTAN "
        f"{ranking['worst_ltan_h']:.2f} h; "
        f"score={ranking['worst_mean_solar_fraction']:.6f}"
    )
    print(
        f"Maior fração solar média: LTAN "
        f"{ranking['best_ltan_h']:.2f} h; "
        f"score={ranking['best_mean_solar_fraction']:.6f}"
    )

    cases = (
        ("Pior Insolacao", "worst_ltan_h"),
        ("Melhor Insolacao", "best_ltan_h"),
    )

    for label, key in cases:
        ltan = ranking[key]
        seasonal_start = ranking["worst_beta_dates"][ltan]

        print(
            f"\nÉpoca selecionada para {label.lower()}: "
            f"{seasonal_start.date()}"
        )

        beta_profile = build_beta_profile(
            seasonal_start,
            ltan,
            MAX_ORBITS,
        )

        result, history_records = simulate(
            pv_input_kw=PV_SCENARIOS_KW[PV_CASE],
            beta_profile_deg=beta_profile,
            dt_s=SIM_DT_S,

# Descrição: Simula o acoplamento eletrotérmico em malha fechada de um nó de 50 kW,

# pré-resfriamento orbital (feedforward) e limitador de rampa de clock contra
# um controlador reativo convencional (bang-bang / histerese).

import csv
import json
import math
from pathlib import Path

import numpy as np

OUTPUT_DIR = Path(__file__).resolve().parent / "saida_4_5"

DT_S = 1.0
T_TOTAL_S = 600.0
T_TARGET_C = 65.0
T_THR_C = 80.0
T_CRIT_C = 85.0

C_NODES = np.array([85.0, 420.0, 1100.0, 3500.0, 12500.0])
R_12 = 0.035
R_23 = 0.025
R_34 = 0.015  # Linearizacao efetiva de primeira ordem no ponto nominal.
R_45 = 0.018

P_IT_BASE_W = 2000.0
P_IT_STEP_W = 40000.0
T_STEP_START_S = 180.0
T_STEP_END_S = 420.0

T_PRECOOL_START_S = 90.0
T_PRECOOL_END_S = 180.0
M_DOT_NOMINAL = 1.2
M_DOT_BOOST = 3.2


def sistema_termico_derivadas(t, T_vec, p_it, m_dot):
    T1, T2, T3, T4, T5 = T_vec
    g34 = (1.0 / R_34) * (m_dot / M_DOT_NOMINAL) ** 0.8
    g12 = 1.0 / R_12
    g23 = 1.0 / R_23
    g45 = 1.0 / R_45

    exp_arg = min(15.0, 0.025 * max(0.0, T1 - 65.0))
    p_leak = 0.05 * p_it * math.exp(exp_arg)
    p_chip = p_it + p_leak

    sigma = 5.670374e-8
    eps_ir = 0.88
    eta_fin = 0.945
    a_rad = 105.0
    q_rad = eps_ir * sigma * a_rad * eta_fin * (
        max(T5 + 273.15, 3.0) ** 4 - 3.0 ** 4
    )

    dT1 = (p_chip - g12 * (T1 - T2)) / C_NODES[0]
    dT2 = (g12 * (T1 - T2) - g23 * (T2 - T3)) / C_NODES[1]
    dT3 = (g23 * (T2 - T3) - g34 * (T3 - T4)) / C_NODES[2]
    dT4 = (g34 * (T3 - T4) - g45 * (T4 - T5)) / C_NODES[3]
    dT5 = (g45 * (T4 - T5) - q_rad) / C_NODES[4]

    return np.array([dT1, dT2, dT3, dT4, dT5])


def run_simulation(control_mode="TEMS"):
    n_steps = int(T_TOTAL_S / DT_S) + 1
    t_arr = np.linspace(0.0, T_TOTAL_S, n_steps)

    T_mat = np.zeros((n_steps, 5))
    T_mat[0] = [65.0, 52.0, 45.0, 38.0, 30.0]

    f_clk_arr = np.zeros(n_steps)
    m_dot_arr = np.zeros(n_steps)
    p_it_arr = np.zeros(n_steps)

    f_clk_current = 1.8

    for k in range(n_steps - 1):
        t = t_arr[k]

        if T_STEP_START_S <= t <= T_STEP_END_S:
            p_it = P_IT_BASE_W + P_IT_STEP_W
        else:
            p_it = P_IT_BASE_W

        if control_mode == "TEMS" and T_mat[k, 0] >= T_THR_C:
            p_it *= 0.80

        p_it_arr[k] = p_it
        T1_current = T_mat[k, 0]

        if control_mode == "TEMS":
            if T_PRECOOL_START_S <= t < T_PRECOOL_END_S:
                m_dot = M_DOT_BOOST
            else:
                m_dot = M_DOT_NOMINAL

            max_slew_per_step = 0.1 * DT_S

            if T1_current >= T_THR_C:
                target_clk = 1.4
            elif T1_current >= T_CRIT_C:
                target_clk = 0.8
            else:
                target_clk = 1.8

            diff = target_clk - f_clk_current
            if abs(diff) <= max_slew_per_step:
                f_clk_current = target_clk
            else:
                f_clk_current += math.copysign(max_slew_per_step, diff)
        else:
            m_dot = M_DOT_NOMINAL
            if T1_current >= T_CRIT_C:
                f_clk_current = 0.8
            elif T1_current <= T_TARGET_C:
                f_clk_current = 1.8
            else:
                f_clk_current = 1.8 if (int(t) % 2 == 0) else 0.8

        f_clk_arr[k] = f_clk_current
        m_dot_arr[k] = m_dot

        derivs = sistema_termico_derivadas(t, T_mat[k], p_it, m_dot)
        T_mat[k + 1] = T_mat[k] + derivs * DT_S

    p_it_arr[-1] = p_it_arr[-2]
    f_clk_arr[-1] = f_clk_arr[-2]
    m_dot_arr[-1] = m_dot_arr[-2]
    T_mat[-1] = T_mat[-2]

    return {
        "t": t_arr,
        "T_juncao": T_mat[:, 0],
        "T_spreader": T_mat[:, 1],
        "T_coldplate": T_mat[:, 2],
        "T_fluido": T_mat[:, 3],
        "T_radiador": T_mat[:, 4],
        "f_clk": f_clk_arr,
        "m_dot": m_dot_arr,
        "p_it": p_it_arr,
    }


def main():
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    print("=" * 80)
    print("SIMULAÇÃO 4.5 — COMPARAÇÃO TEMS-NMPC vs. CONTROLE REATIVO")
    print("=" * 80)

    res_tems = run_simulation("TEMS")
    res_reactive = run_simulation("Reactive")

    tj_max_tems = np.max(res_tems["T_juncao"])
    tj_max_reactive = np.max(res_reactive["T_juncao"])

    csv_path = OUTPUT_DIR / "simulacao_4_5_comparacao_resumo.csv"

def cycles_per_day(period_s):
    if period_s <= 0:
        raise ValueError("O período orbital precisa ser positivo.")
    return 86400.0 / period_s


def strain_range(delta_tj_c):
    """
    Aproximação geométrica comum aos modelos comparados:

    Δγ = F_conf * L_DNP * Δα * ΔT_s / h_s
    ΔT_s = K_att * ΔT_j

    Esta estimativa não substitui análise mecânica da montagem.
    """
    if delta_tj_c <= 0:
        raise ValueError("ΔT_j precisa ser positivo.")

    l_dnp = 0.5 * math.hypot(L_X_M, L_Y_M)
    delta_ts_c = K_ATT * delta_tj_c
    gamma = (
        F_CONF
        * l_dnp
        * DELTA_ALPHA_K_INV
        * delta_ts_c
        / H_S_M
    )
    return l_dnp, delta_ts_c, gamma


def engelmaier_exponent(temp_c, frequency_cpd, coefficients):
    return (
        coefficients["a"]
        + coefficients["b"] * temp_c
        + coefficients["d"] * math.log1p(frequency_cpd)
    )


def cycles_to_failure(gamma, two_eps_f, exponent):
    """
    Forma comparativa usada nos scripts recebidos:

    Nf = 0.5 * (γ / (2ε'f)) ** (1/c)

    O parâmetro 2ε'f e a forma de estimar γ precisam de validação
    para o material e o empilhamento reais.
    """
    if gamma <= 0 or two_eps_f <= 0:
        raise ValueError("γ e 2ε'f precisam ser positivos.")
    if exponent >= 0:
        raise ValueError("O expoente c precisa ser negativo.")

    return 0.5 * (gamma / two_eps_f) ** (1.0 / exponent)

# ---------------------------------------------------------------------
# SSPC: envelopes condicionais e energia idealizada de desmagnetização
# ---------------------------------------------------------------------

def sspc_analysis():
    if ETA_DAB <= 0 or V_TVS_CLAMP_V <= V_BUS_V:
        raise ValueError("Eficiência ou tensão de clamp inválida.")

    i_nom = P_NODE_W / V_BUS_V
    i_limit = I_TRIP_MULTIPLIER * i_nom

    # Limite retangular condicional, se tensão e corrente permanecerem
    # limitadas pelos valores assumidos durante todo o tempo de trip.
    e_switch_rectangular_j = (
        V_TVS_CLAMP_V * i_limit * T_TRIP_S
    )

    # Energia inicialmente armazenada na indutância do chicote.
    e_inductor_j = 0.5 * L_HARNESS_H * i_limit**2

    # Modelo idealizado de clamp, assumindo corrente que decresce
    # linearmente por (Vclamp - Vbus) sobre a indutância.
    delta_v = V_TVS_CLAMP_V - V_BUS_V
    t_demag_s = L_HARNESS_H * i_limit / delta_v

    # Energia absorvida pelo TVS no modelo idealizado.
    # Inclui energia que a fonte pode fornecer durante a desmagnetização.
    e_tvs_ideal_j = (
        e_inductor_j * V_TVS_CLAMP_V / delta_v
    )
    e_source_during_demag_j = (
        0.5 * V_BUS_V * i_limit * t_demag_s
    )

    return {
        "i_nom_a": i_nom,
        "i_limit_assumed_a": i_limit,
        "trip_time_us_assumed": T_TRIP_S * 1e6,
        "switch_energy_rectangular_bound_j": e_switch_rectangular_j,

# ---------------------------------------------------------------------
# PUE orbital adaptado: razão entre energias integradas
# ---------------------------------------------------------------------

def pue_for_case(orbit_case, tems_enabled):
    period = orbit_case["period_s"]
    eclipse = orbit_case["eclipse_s"]

    if not (0 <= eclipse < period):
        raise ValueError("Duração de eclipse incompatível com o período.")

    sun_time = period - eclipse
    eclipse_fraction = eclipse / period
    sun_fraction = sun_time / period

    ti_sun = orbit_case["ti_sun_w"]

    if tems_enabled:
        ti_eclipse = orbit_case["ti_eclipse_tems_w"]
        control_name = "TEMS"
    else:
        ti_eclipse = orbit_case["ti_eclipse_no_tems_w"]
        control_name = "Sem TEMS"

    # Energias úteis de TI em uma órbita.
    e_it_sun = ti_sun * sun_time
    e_it_eclipse = ti_eclipse * eclipse
    e_it_total = e_it_sun + e_it_eclipse

    # Perdas DAB: a potência útil de TI é considerada saída do DAB.
    e_dab_loss = (
        ti_sun * (1.0 / ETA_DAB - 1.0) * sun_time
        + ti_eclipse * (1.0 / ETA_DAB - 1.0) * eclipse
    )

    # Cargas variáveis por fase orbital.
    e_base = (
        P_BASE_SUN_W * sun_time
        + P_BASE_ECLIPSE_W * eclipse
    )
    e_heater = (
        P_BATTERY_HEATER_SUN_W * sun_time
        + P_BATTERY_HEATER_ECLIPSE_W * eclipse
    )

    # Cargas elétricas auxiliares e OISL.
    p_aux_electrical = (
        P_PUMP_W + P_AVIONICS_W + P_OISL_ELECTRICAL_W
    )
    e_aux = p_aux_electrical * period

    # Perda BESS como hipótese independente. Revisar a fronteira e
    # evitar somá-la se já estiver embutida em P_BASE ou noutra parcela.
    e_bess_loss = P_BESS_LOSS_W * period

    e_total = (
        e_it_total
        + e_dab_loss
        + e_base
        + e_heater
        + e_aux
        + e_bess_loss
    )

    if e_it_total <= 0:
        raise ValueError("Energia de TI precisa ser positiva.")

    pue = e_total / e_it_total
    tops_peak = N_DIES * TOPS_PER_DIE

    # Indicador de pico nominal; não é throughput medido.
    tops_per_w = tops_peak / (e_total / period)

    return {
        "orbit": orbit_case["orbit"],
        "control": control_name,
        "period_s": period,
        "eclipse_s": eclipse,
        "sun_fraction": sun_fraction,
        "eclipse_fraction": eclipse_fraction,
        "ti_sun_w": ti_sun,
        "ti_eclipse_w": ti_eclipse,
        "p_base_sun_w_provisional": P_BASE_SUN_W,
        "p_base_eclipse_w_provisional": P_BASE_ECLIPSE_W,
        "heater_sun_w_assumed": P_BATTERY_HEATER_SUN_W,
        "heater_eclipse_w_assumed": P_BATTERY_HEATER_ECLIPSE_W,
        "oisl_electrical_w": P_OISL_ELECTRICAL_W,
        "oisl_internal_heat_w_separate": Q_OISL_INTERNAL_HEAT_W,
        "p_aux_electrical_w": p_aux_electrical,
        "bess_loss_w_assumed": P_BESS_LOSS_W,
        "eta_dab_assumed": ETA_DAB,
        "it_energy_j": e_it_total,
        "dab_loss_energy_j": e_dab_loss,
        "base_energy_j": e_base,
        "heater_energy_j": e_heater,
        "auxiliary_energy_j": e_aux,
        "bess_loss_energy_j": e_bess_loss,
        "total_boundary_energy_j": e_total,
        "pue_orbital_adapted": pue,
        "tops_peak_total": tops_peak,
        "tops_per_w_nominal_peak": tops_per_w,
        "warning": (
            "Indicador adaptado, nao alegacao de conformidade ISO. "
            "P_BASE e aquecimento sao configuraveis; verificar "
            "sobreposicao antes de interpretar."
        ),
    }

"""
Simulacao Numerica de Transiente de Curto-Circuito em Barramento HVDC (400 V)

Determinacao dos limites criticos de abertura da chave SSPC (SiC MOSFET):
1. Tempo critico de desarme (t_trip) para atingir I^2t = 1500 A^2s.
2. Tempo de desarme que atinge V_spike = 1200 V.
3. Tempo minimo de extincao (soft turn-off) para evitar V_spike > 1200 V.

ATENCAO:
- O tempo de extincao calculado na secao 3 usa uma estimativa analitica:
  V_spike ~= V_bus + L * I / tau_ext.
- Os resultados dependem das hipoteses e limitacoes do modelo numerico.
"""

import numpy as np
import matplotlib.pyplot as plt


# ============================================================
# MODELO TRANSITORIO
# ============================================================

def simular_curto(
    t_trip,
    L_line,
    C_bus,
    R_line=14.0e-3,
    R_sc=1.0e-3,
    V_bus=400.0,
    V_clamp=550.0,
    clamp_ativo=True,
    tau_ext=0.5e-6
):
    dt = 0.01e-6  # 10 ns
    t_end = max(150e-6, t_trip + 40e-6)
    t = np.arange(0.0, t_end, dt)

    i_curr = 0.0
    v_curr = V_bus
    i2t = 0.0
    v_sw_max = 0.0
    i_peak = 0.0

    for tk in t:

        if tk < t_trip:
            di_dt = (
                v_curr - i_curr * (R_line + R_sc)
            ) / L_line

            dv_dt = -i_curr / C_bus

            i_curr = i_curr + di_dt * dt
            v_curr = v_curr + dv_dt * dt

            v_sw = i_curr * 0.010  # Modelo original de Rds_on

        else:
            if clamp_ativo:
                di_dt = -(
                    V_clamp - v_curr + i_curr * R_line
                ) / L_line

                dv_dt = -i_curr / C_bus

                i_curr = max(0.0, i_curr + di_dt * dt)
                v_curr = v_curr + dv_dt * dt

                v_sw = V_clamp if i_curr > 0.0 else v_curr

            else:
                di_dt = -i_curr / tau_ext
                v_indutivo = abs(L_line * di_dt)

                i_curr = max(0.0, i_curr + di_dt * dt)

                dv_dt = -i_curr / C_bus
                v_curr = v_curr + dv_dt * dt

                v_sw = v_curr + v_indutivo

        i2t += (i_curr ** 2) * dt

        if i_curr > i_peak:
            i_peak = i_curr

        if v_sw > v_sw_max:
            v_sw_max = v_sw

        if tk > t_trip and i_curr <= 0.0:
            break

    return i_peak, i2t, v_sw_max

def primeiro_cruzamento(valores, limite, tempos):
    """Retorna o tempo do primeiro cruzamento, ou None."""
    indices = np.where(valores >= limite)[0]

    if len(indices) == 0:
        return None

    idx = indices[0]

    if idx == 0:
        return float(tempos[0])

    # Interpolacao linear para melhorar a estimativa temporal
    return float(
        np.interp(
            limite,
            [valores[idx - 1], valores[idx]],
            [tempos[idx - 1], tempos[idx]]
        )
    )

# Estimativa analitica:
# V_spike ~= V_bus + L * I / tau_ext
#
# Portanto:
# tau_ext >= L * I / (V_limite - V_bus)

delta_V_max = V_SPIKE_LIMITE - V_BUS


def calcular_tau_min(L_line, i_corrente):
    """Calcula tau minimo estimado, em segundos."""
    if delta_V_max <= 0.0:
        raise ValueError(
            "O limite de tensao deve ser superior a V_BUS."
        )

    return L_line * i_corrente / delta_V_max

import numpy as np


def executar_bateria_100_orbitas():
    print("=" * 111)
    print("INICIANDO SIMULAÇÃO 3.3 EXPANDIDA: BALANÇO ELETROTÉRMICO DO BESS EM 100 ÓRBITAS")
    print("=" * 111 + "\n")

    Cn = 22.52
    Vuv = 324.0
    Vov = 442.8
    Rp1 = 18e-3
    Cp1 = 3200.0
    Rp2 = 24e-3
    Cp2 = 22000.0
    R0_ref = 85e-3
    Ea = 0.32 * 1.602176634e-19
    kB = 1.380649e-23
    Tref_K = 293.15

    Cth = 54360.0
    Rth_cond = 0.14
    Tch = 15.0
    Pheater = 300.0
    Ton = 10.0
    Toff = 15.0

    dt = 2.0
    Torb = 5926.0
    tecl = 1128.0
    nOrb = 100
    steps_per_orb = int(np.floor(Torb / dt))
    N = nOrb * steps_per_orb + 1

    soc = np.zeros(N)
    vt = np.zeros(N)
    Tb = np.zeros(N)
    Pj = np.zeros(N)

    soc[0] = 0.90
    Tb[0] = 20.0
    vp1 = 0.0
    vp2 = 0.0
    a1 = np.exp(-dt / (Rp1 * Cp1))
    a2 = np.exp(-dt / (Rp2 * Cp2))

    alvos = [1, 5, 10, 20, 50, 100]

    print(f"{'Orbita':<8} | {'Tempo':<8} | {'SoC Min/Max (%)':<16} | {'Tensao Min/Max(V)':<16} | {'T_bat (Min/Med/Max) C':<24} | {'Pj_med (W)':<10} | {'Teq_est C':<10}")
    print("-" * 111)

    for k in range(N - 1):
        t_now = k * dt
        in_eclipse = (t_now % Torb) <= tecl

        if in_eclipse:
            I = (18000.0 + 560.0) / 380.0
            etaC = 1.0
        else:
            I = -25.0
            etaC = 0.995

        R0 = R0_ref * np.exp((Ea / kB) * (1.0 / (Tb[k] + 273.15) - 1.0 / Tref_K))
        vp1 = a1 * vp1 + Rp1 * (1.0 - a1) * I
        vp2 = a2 * vp2 + Rp2 * (1.0 - a2) * I

        soc[k + 1] = min(1.0, max(0.0, soc[k] - I * etaC * dt / (Cn * 3600.0)))
        Voc = 108.0 * (3.0 + 1.1 * soc[k + 1])
        Vcand = Voc - I * R0 - vp1 - vp2

        if I < 0.0 and Vcand > Vov:
            I = max(-25.0, min(0.0, -(Vov - Voc + vp1 + vp2) / R0))
            Vcand = Vov
        vt[k + 1] = Vcand

        Pj[k] = (I**2) * R0 + (vp1**2) / Rp1 + (vp2**2) / Rp2
        if Tb[k] < Ton:
            Ph = Pheater
        elif Tb[k] < Toff:
            Ph = Pheater * (Toff - Tb[k]) / (Toff - Ton)
        else:
            Ph = 0.0

        q_out = (Tb[k] - Tch) / Rth_cond
        Tb[k + 1] = Tb[k] + (Pj[k] + Ph - q_out) * dt / Cth

        if (k + 1) % steps_per_orb == 0:
            orb_atual = (k + 1) // steps_per_orb
            if orb_atual in alvos:
                idx_start = (orb_atual - 1) * steps_per_orb
                idx_end = k + 2

                soc_min = np.min(soc[idx_start:idx_end]) * 100.0
                soc_max = np.max(soc[idx_start:idx_end]) * 100.0
                vt_min = np.min(vt[idx_start:idx_end])
                vt_max = np.max(vt[idx_start:idx_end])
                tb_min = np.min(Tb[idx_start:idx_end])
                tb_med = np.mean(Tb[idx_start:idx_end])
                tb_max = np.max(Tb[idx_start:idx_end])
                pj_med = np.mean(Pj[idx_start:k+1])
                teq_est = Tch + pj_med * Rth_cond
                t_horas = ((k + 1) * dt) / 3600.0

                print(f"{orb_atual:<8} | {t_horas:6.1f} h | [{soc_min:5.2f}, {soc_max:6.2f}]% | [{vt_min:5.1f}, {vt_max:5.1f}] V | [{tb_min:5.2f}, {tb_med:5.2f}, {tb_max:5.2f}] C | {pj_med:8.1f} W | {teq_est:7.2f} C")

    print("=" * 111)
    print("SIMULAÇÃO DE 100 ÓRBITAS CONCLUÍDA COM SUCESSO.")
    print("=" * 111)


if __name__ == '__main__':
    executar_bateria_100_orbitas()
