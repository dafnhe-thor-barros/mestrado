    return np.array([(pc-g12*(T1-T2))/C[0],(g12*(T1-T2)-g23*(T2-T3))/C[1],(g23*(T2-T3)-g34*(T3-T4))/C[2],(g34*(T3-T4)-g45*(T4-T5))/C[3],(g45*(T4-T5)-qr)/C[4]])

    return (eit+edab+pbase*period+((200+350+440)*period if aux else 0)+bess*period)/eit
