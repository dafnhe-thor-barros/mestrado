WGS84_INV_F = 298.257223563           # [FONTE] idem
WGS84_GM = 3.986004418e14             # [FONTE] idem [m3/s2]
WGS84_J2 = 1.082629821313e-3          # [FONTE] idem, "J2geo" (tabela lida via texto extraido; pagina nao vista)
WGS84_B_M = WGS84_A_M * (1.0 - 1.0 / WGS84_INV_F)   # [CALCULADO]; NGA lista b = 6356752.3142 m

R_SUN_M = 6.957e8                     # [FONTE] IAU 2015 Res. B3 (lida)

S_NOMINAL = 1361.0                    # [FONTE] IAU 2015 Res. B3 (lida); Kopp&Lean 2011: 1360.8+-0.5 (lida)
SUN_RATE_RAD_S = 2 * math.pi / (365.2422 * 86400.0)   # [NAO VERIFICADO] ano tropico 365.2422 d lembrado de memoria; o primer GSFC contem os tokens "0.9856" e "98.2", mas o slide/contexto NAO foi visto

def rad(x): return math.radians(x)
def deg(x): return math.degrees(x)

# ---------------- Tempo e Sol ----------------
def jd_from_datetime(dt):
    """JD = 2440587.5 + t_unix/86400 (UTC). Verificado por teste: 2000-01-01 12:00Z -> 2451545.0."""
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return 2440587.5 + dt.timestamp() / 86400.0

def sun_usno(jd):
    """Algoritmo USNO 'Computing Approximate Solar Coordinates' [FONTE, lida]: ~1 arcmin em +-2 seculos de 2000."""
    D = jd - 2451545.0
    g = rad((357.529 + 0.98560028 * D) % 360.0)
    q = (280.459 + 0.98564736 * D) % 360.0
    L = rad((q + 1.915 * math.sin(g) + 0.020 * math.sin(2 * g)) % 360.0)
    R = 1.00014 - 0.01671 * math.cos(g) - 0.00014 * math.cos(2 * g)
    e = rad(23.439 - 0.00000036 * D)
    ra = math.atan2(math.cos(e) * math.sin(L), math.cos(L))
    dec = math.asin(math.sin(e) * math.sin(L))
    eqt_h = (q / 15.0 - deg(ra) / 15.0 + 12.0) % 24.0 - 12.0   # EqT = q/15 - RA (USNO)
    vec = np.array([math.cos(dec) * math.cos(ra), math.cos(dec) * math.sin(ra), math.sin(dec)])
    return {"vec": vec, "ra_deg": deg(ra), "dec_deg": deg(dec), "R_au": R, "q_deg": q, "eqt_min": eqt_h * 60.0}

# ---------------- Orbita ----------------
def kepler_period_s(alt_m):
    a = WGS84_A_M + alt_m
    return 2 * math.pi * math.sqrt(a ** 3 / WGS84_GM)

def sso_inclination_deg(alt_m):
    """cos i = -Omega_sun / (1.5 J2 (Re/a)^2 n)  com dOmega/dt = -1.5 J2 (Re/p)^2 n cos i, e=0.
    [NAO VERIFICADO em fonte lida] A formula so foi conferida por CONSISTENCIA CRUZADA com o propagador J2 numerico (test_j2_taxa_secular_vs_propagador).
    'a' e interpretado como semi-eixo MEDIO."""
    a = WGS84_A_M + alt_m
    n = math.sqrt(WGS84_GM / a ** 3)
    return deg(math.acos(-SUN_RATE_RAD_S / (1.5 * WGS84_J2 * (WGS84_A_M / a) ** 2 * n)))

def raan_from_ltan_deg(ltan_h, q_deg):
    """LTAN em tempo solar MEDIO local: Omega = alpha_sol_medio + 15 deg*(LTAN-12h); alpha_sol_medio = q (USNO). [CALCULADO]"""
    return (q_deg + 15.0 * (ltan_h - 12.0)) % 360.0

def orbit_frame(raan, inc):
    cO, sO, ci, si = math.cos(raan), math.sin(raan), math.cos(inc), math.sin(inc)
    P = np.array([cO, sO, 0.0]); Q = np.array([-sO * ci, cO * ci, si])
    return P, Q, np.cross(P, Q)

def orbit_geometry(dt, ltan_h, alt_m, inc_deg=None, raan_deg=None, shift_s=0.0, const_sun_dist=False):
    jd = jd_from_datetime(dt) + shift_s / 86400.0
    s = sun_usno(jd)
    inc = sso_inclination_deg(alt_m) if inc_deg is None else inc_deg
    raan = raan_from_ltan_deg(ltan_h, s["q_deg"]) if raan_deg is None else raan_deg
    P, Q, h = orbit_frame(rad(raan), rad(inc))
    beta = deg(math.asin(max(-1.0, min(1.0, float(h @ s["vec"])))))
    R_au = 1.0 if const_sun_dist else s["R_au"]
    rs = math.asin(R_SUN_M / (R_au * AU_M))
    return {"beta_deg": beta, "raan_deg": raan, "inc_deg": inc, "sun": s["vec"], "rs_rad": rs, "T_s": kepler_period_s(alt_m), "q_deg": s["q_deg"]}

def raan_for_beta0(dt, alt_m, inc_deg):
    s = sun_usno(jd_from_datetime(dt))["vec"]
    best = (1e9, 0.0)
    for k in range(7200):
        o = k * 0.05
        _, _, h = orbit_frame(rad(o), rad(inc_deg))
        v = abs(float(h @ s))
        if v < best[0]: best = (v, o)
    return best[1]

# ---------------- Eclipse ----------------
def sunlit_fraction(g, rs, re, legacy=False, clip=True):
    """Fracao do disco solar visivel; g = separacao angular Sol-centro da Terra. Interseccao de 2 discos (lente planar).
    Raio: A=rs^2 acos(.)+re^2 acos(.)-0.5 sqrt((-d+rs+re)(d+rs-re)(d-rs+re)(d+rs+re)). legacy=True reproduz o 3o fator errado."""
    g = np.atleast_1d(np.asarray(g, float)); out = np.ones_like(g)
    out[g <= re - rs] = 0.0
    pen = (g > re - rs) & (g < re + rs)
    s = g[pen]
    a1 = np.clip((s * s + rs * rs - re * re) / (2 * s * rs), -1, 1)
    a2 = np.clip((s * s + re * re - rs * rs) / (2 * s * re), -1, 1)
    f3 = (s - re + rs) if legacy else (s - rs + re)
    rad_ = np.maximum(0.0, (-s + rs + re) * (s + rs - re) * f3 * (s + re + rs))
    ov = rs * rs * np.arccos(a1) + re * re * np.arccos(a2) - 0.5 * np.sqrt(rad_)
    out[pen] = 1.0 - ov / (math.pi * rs * rs)
    return np.clip(out, 0.0, 1.0) if clip else out

def eclipse_cyl_s(beta_deg, alt_m):
    """Sombra cilindrica (Sol pontual): t = (T/pi) acos( sqrt(1-(Re/r)^2)/cos(beta) ). [CALCULADO]"""
    r = WGS84_A_M + alt_m
    x = math.sqrt(1.0 - (WGS84_A_M / r) ** 2) / math.cos(rad(beta_deg))
    return kepler_period_s(alt_m) / math.pi * math.acos(x) if x < 1.0 else 0.0

def eclipse_cone_s(beta_deg, alt_m, rs, n=20000, legacy=False):
    """Eclipse equivalente por orbita = integral de (1-fracao_solar) dt; cos(gamma) = -cos(beta) cos(theta)."""
    r = WGS84_A_M + alt_m; T = kepler_period_s(alt_m); rho = math.asin(WGS84_A_M / r)
    th = (np.arange(n) + 0.5) * 2 * math.pi / n
    g = np.arccos(np.clip(-math.cos(rad(beta_deg)) * np.cos(th), -1, 1))
    return float((1.0 - sunlit_fraction(g, rs, rho, legacy)).sum() * T / n)

def eclipse_ray_s(raan_deg, inc_deg, sun_vec, alt_m, a_m=WGS84_A_M, b_m=WGS84_B_M, n=100000):
    """Teste independente por raio contra elipsoide (Sol pontual, raios paralelos): z' = z a/b transforma em esfera."""
    r = WGS84_A_M + alt_m; T = kepler_period_s(alt_m)
    P, Q, _ = orbit_frame(rad(raan_deg), rad(inc_deg))
    u = (np.arange(n) + 0.5) * 2 * math.pi / n
    pos = r * (np.cos(u)[:, None] * P + np.sin(u)[:, None] * Q)
    k = a_m / b_m
    p = pos.copy(); p[:, 2] *= k
    d = np.array(sun_vec, float); d[2] *= k
    A = float(d @ d); B = 2.0 * (p @ d); C = (p * p).sum(1) - a_m ** 2
    return float(((B < 0) & (B * B - 4 * A * C > 0)).sum() * T / n)

# ---------------- Fator de visao da Terra ----------------
def view_factor_closed(alt_m, phi_deg, legacy=False):
    H = (WGS84_A_M + alt_m) / WGS84_A_M; thm = math.asin(1.0 / H); root = math.sqrt(H * H - 1.0); phi = rad(phi_deg)
    cl = lambda v, lo, hi: max(lo, min(hi, v))
    if phi <= math.pi / 2 - thm: return cl(math.cos(phi) / H ** 2, 0.0, 1.0)
    if phi <= math.pi / 2 + thm:
        sp = math.sin(phi)
        a1 = cl(root / (H * sp), -1, 1)
        a2 = cl((1.0 if legacy else -1.0) * root * math.cos(phi) / sp, -1, 1)
        rd = max(0.0, 1.0 - H * H * math.cos(phi) ** 2)
        f = 0.5 - math.asin(a1) / math.pi + math.cos(phi) / (math.pi * H * H) * math.acos(a2) - root * math.sqrt(rd) / (math.pi * H * H)
        return cl(f, 0.0, 1.0)
    return 0.0

def view_factor_quad(alt_m, phi_deg, na=1200, nz=1200):
    """F = (1/pi) Int_calota max(0, cos psi) dOmega (Terra lambertiana uniforme: 1a aproximacao SP-8067, via resumo)."""
    H = (WGS84_A_M + alt_m) / WGS84_A_M; thm = math.asin(1.0 / H); phi = rad(phi_deg)
    al = (np.arange(na) + 0.5) * thm / na; ze = (np.arange(nz) + 0.5) * 2 * math.pi / nz
    A, Z = np.meshgrid(al, ze, indexing="ij")
    c = np.sin(A) * np.cos(Z) * math.sin(phi) + np.cos(A) * math.cos(phi)
    return float((np.maximum(c, 0.0) * np.sin(A)).sum() * (thm / na) * (2 * math.pi / nz) / math.pi)

# ---------------- Atitude do radiador e fluxos ambientais ----------------
def radiator_env(theta, beta_deg, alt_m, law="s_cross_r", legacy_vf=False, albedo=ALBEDO, q_ir=Q_IR, eps=EPS_RAD, alpha=ALPHA_RAD, G=S_NOMINAL):
    """Sistema do legado: r=(cos th, sin th, 0), s=(cos b, 0, sin b). Leis: 's_cross_r' (opcao 1, n = s x r/|.|, phi=90 sempre),
    'inertial_fixed' (opcao 2, n = h_perp_s fixo no inercial), 'legacy' (n=-s). Por face (dupla face: soma F(phi)+F(180-phi) se phi!=90).
    Albedo no ponto subsatelite: q = alpha G a F mu, mu=max(0, r.s) [HIPOTESE DECLARADA, herdada do legado]."""
    b = rad(beta_deg)
    r = np.array([math.cos(theta), math.sin(theta), 0.0]); s = np.array([math.cos(b), 0.0, math.sin(b)]); h = np.array([0.0, 0.0, 1.0])
    if law == "s_cross_r":
        n = np.cross(s, r); nn = np.linalg.norm(n)
        n = h if nn < 1e-9 else n / nn      # degenerado so se r || s (beta ~ 0): usa normal da orbita
    elif law == "inertial_fixed":
        n = h - float(h @ s) * s; n = n / np.linalg.norm(n)
    else:
        n = -s
    phi = deg(math.acos(max(-1.0, min(1.0, float(-n @ r)))))
    F = view_factor_closed(alt_m, phi, legacy=legacy_vf)
    mu = max(0.0, float(r @ s))
    return {"phi_deg": phi, "F": F, "q_ir": eps * q_ir * F, "q_alb": alpha * G * albedo * F * mu,
            "q_sol": alpha * G * max(0.0, float(n @ s)), "n": n}

# ---------------- Propagador J2 (verificacao) ----------------
def _acc_j2(t, y):
    r = y[:3]; rn = float(np.linalg.norm(r)); k = 1.5 * WGS84_J2 * (WGS84_A_M / rn) ** 2; z2 = (r[2] / rn) ** 2
    f = -WGS84_GM / rn ** 3
    return np.concatenate([y[3:], [f * r[0] * (1 - k * (5 * z2 - 1)), f * r[1] * (1 - k * (5 * z2 - 1)), f * r[2] * (1 - k * (5 * z2 - 3))]])

def propagate_j2(raan_deg, inc_deg, alt_m, t_end_s, step_s):
    r0 = WGS84_A_M + alt_m; P, Q, _ = orbit_frame(rad(raan_deg), rad(inc_deg))
    y0 = np.concatenate([r0 * P, math.sqrt(WGS84_GM / r0) * Q]); ts = np.arange(0.0, t_end_s + step_s, step_s)
    sol = solve_ivp(_acc_j2, (0.0, ts[-1]), y0, method="DOP853", t_eval=ts, rtol=1e-10, atol=1e-4)
    return ts, sol.y[:3].T, sol.y[3:].T

def j2_drift_deg_day(ts, R, V):
    hv = np.cross(R, V); om = np.unwrap(np.arctan2(hv[:, 0], -hv[:, 1]))
    return deg(np.polyfit(ts, om, 1)[0]) * 86400.0

def secular_raan_deg_day(alt_m, inc_deg):
    a = WGS84_A_M + alt_m; n = math.sqrt(WGS84_GM / a ** 3)
    return deg(-1.5 * WGS84_J2 * (WGS84_A_M / a) ** 2 * n * math.cos(rad(inc_deg))) * 86400.0

def test_lente_vs_integracao_esferica():
    rs = math.asin(R_SUN_M / AU_M); re = math.asin(WGS84_A_M / (WGS84_A_M + 700e3)); worst = 0.0
    xs = np.linspace(-rs, rs, 301); X, Y = np.meshgrid(xs, xs); r = np.hypot(X, Y); m = r <= rs; ph = np.arctan2(Y, X)
    vx = np.sin(r) * np.cos(ph); vz = np.cos(r)
    for k in (-0.9, -0.5, 0.0, 0.5, 0.9):
        g = re + k * rs
        ex = float(((np.arccos(np.clip(vx * math.sin(g) + vz * math.cos(g), -1, 1)) > re) & m).sum() / m.sum())
        worst = max(worst, abs(float(sunlit_fraction(g, rs, re)[0]) - ex))
    leg = float(sunlit_fraction(re, rs, re, legacy=True, clip=False)[0]); MEAS["lente_max_err"] = worst; MEAS["lente_legado_k0_sem_clamp"] = leg
    assert worst < 4e-3, worst
    assert leg < -10.0, leg

def _legacy_beta(dt, ltan, inc=98.2):
    doy = dt.timetuple().tm_yday; fd = doy + (0.0 - 12.0) / 24.0
    d = rad(23.44 * math.sin(rad(360.0 * (fd - 81.0) / 365.0))); dw = rad(15.0 * (ltan - 12.0)); i = rad(inc)
    return deg(math.asin(max(-1, min(1, math.sin(i) * math.cos(d) * math.sin(dw) + math.cos(i) * math.sin(d)))))

def _ecl_day(beta, alt, rs, kind, n=20000):
    if abs(beta) > 64.9: return 0.0
    return eclipse_cyl_s(beta, alt) if kind == "cyl" else eclipse_cone_s(beta, alt, rs, n=n)

    a = np.abs(np.array(arr)); return {"max_abs": float(a.max()), "dia_max": int(a.argmax()), "n_dias_gt_limiar": int((a > thr).sum())}
